Example 1 — Grade Classifier (if / else if chain)

Demonstrates multi-branch decision making with boundary checks and error validation using if, else if, and else.

01_ifelse.c
#include <stdio.h>

int main(void) {
    int scores[] = {95, 82, 74, 61, 48, -5, 105};
    int count = sizeof(scores) / sizeof(scores[0]);

    for (int i = 0; i < count; i++) {
        int score = scores[i];
        printf("Score: %3d -> ", score);

        if (score < 0 || score > 100) {
            printf("Invalid score!\n");
        } else if (score >= 90) {
            printf("Grade A (Excellent)\n");
        } else if (score >= 80) {
            printf("Grade B (Good)\n");
        } else if (score >= 70) {
            printf("Grade C (Satisfactory)\n");
        } else if (score >= 60) {
            printf("Grade D (Pass)\n");
        } else {
            printf("Grade F (Fail)\n");
        }
    }
    return 0;
}
▶

Output:

Score: 74 -> Grade C (Satisfactory)
Score: 61 -> Grade D (Pass)
Score: 48 -> Grade F (Fail)
Score: -5 -> Invalid score!
Score: 105 -> Invalid score!

Example 2 — Day of Week Printer (switch with fall-through)

Demonstrates switch, case, default, and deliberate case fall-through to classify weekdays and weekends.

02_switch.c
#include <stdio.h>

int main(void) {
    int test_days[] = {1, 3, 5, 6, 7, 9};
    int count = sizeof(test_days) / sizeof(test_days[0]);

    for (int i = 0; i < count; i++) {
        int day = test_days[i];
        printf("Day %d: ", day);

        /* Print day name */
        switch (day) {
            case 1: printf("Monday");    break;
            case 2: printf("Tuesday");   break;
            case 3: printf("Wednesday"); break;
            case 4: printf("Thursday");  break;
            case 5: printf("Friday");    break;
            case 6: printf("Saturday");  break;
            case 7: printf("Sunday");    break;
            default:
                printf("Unknown Day\n");
                continue; /* Skip category check for invalid days */
        }

        /* Classify type using deliberate fall-through */
        switch (day) {
            case 1:
            case 2:
            case 3:
            case 4:
            case 5:
                /* Deliberate fall-through: cases 1-5 share same handler */
                printf(" (Weekday - Time to work)\n");
                break;
            case 6:
            case 7:
                /* Deliberate fall-through: weekend days */
                printf(" (Weekend - Time to relax)\n");
                break;
        }
    }
    return 0;
}
▶
Output:
Day 1: Monday (Weekday - Time to work)
Day 3: Wednesday (Weekday - Time to work)
Day 5: Friday (Weekday - Time to work)
Day 6: Saturday (Weekend - Time to relax)
Day 7: Sunday (Weekend - Time to relax)
Day 9: Unknown Day

Example 3 — Collatz Conjecture (while loop)

Computes the sequence of numbers reaching 1 where n = n/2 if even, and n = 3n + 1 if odd.

03_while.c
#include <stdio.h>

int main(void) {
    long long n = 27;
    int steps = 0;

    printf("Collatz sequence for n = %lld:\n", n);
    printf("%lld", n);

    while (n > 1) {
        if (n % 2 == 0) {
            n = n / 2;
        } else {
            n = 3 * n + 1;
        }
        printf(" -> %lld", n);
        steps++;

        /* Wrap output line every 8 values for clean formatting */
        if (steps % 8 == 0) {
            printf("\n");
        }
    }

    printf("\n\nReached 1 in %d total steps.\n", steps);
    return 0;
}
▶
Output Summary:
Collatz sequence for n = 27:
27 -> 82 -> 41 -> 124 -> 62 -> 31 -> 94 -> 47
... -> 1

Reached 1 in 111 total steps.

Example 4 — Input Validation Loop (do-while loop)

Simulates re-prompting for user input until a valid value between 1 and 100 is provided.

04_dowhile.c
#include <stdio.h>

int main(void) {
    /* Simulated stream of user inputs: negative, too large, zero, valid */
    int simulated_inputs[] = { -12, 145, 0, 75 };
    int attempt = 0;
    int value;

    printf("--- Simulated Score Entry ---\n");

    do {
        value = simulated_inputs[attempt];
        printf("Attempt %d: User entered %d\n", attempt + 1, value);

        if (value < 1 || value > 100) {
            printf("  [ERROR] Score must be between 1 and 100. Please try again.\n");
        }
        attempt++;
    } while (value < 1 || value > 100);

    printf("Success! Valid score accepted: %d\n", value);
    return 0;
}
▶
Output:

Attempt 1: User entered -12
  [ERROR] Score must be between 1 and 100. Please try again.
Attempt 2: User entered 145
  [ERROR] Score must be between 1 and 100. Please try again.
Attempt 3: User entered 0
  [ERROR] Score must be between 1 and 100. Please try again.
Attempt 4: User entered 75
Success! Valid score accepted: 75

Example 5 — Multiplication Table (nested for loops)

Generates a clean 10×10 multiplication grid with formatted column widths using nested for loops.

05_for.c
#include <stdio.h>

int main(void) {
    printf("     ");
    for (int col = 1; col <= 10; col++) {
        printf("%4d", col);
    }
    printf("\n   +----------------------------------------\n");

    for (int row = 1; row <= 10; row++) {
        printf("%2d |", row);
        for (int col = 1; col <= 10; col++) {
            printf("%4d", row * col);
        }
        printf("\n");
    }
    return 0;
}

Example 6 — Search & Filter (break and continue)

Uses break for early loop termination when finding a prime, and continue to skip even numbers.

06_break_continue.c
#include <stdio.h>
#include <stdbool.h>

int main(void) {
    /* Part A: Using break to find the first prime number >= 50 */
    int first_prime = -1;
    for (int n = 50; n <= 100; n++) {
        bool is_prime = true;
        for (int d = 2; d * d <= n; d++) {
            if (n % d == 0) {
                is_prime = false;
                break; /* Not prime, stop checking divisors for this n */
            }
        }
        if (is_prime) {
            first_prime = n;
            break; /* Found first prime >= 50, terminate search loop */
        }
    }
    printf("First prime >= 50 is: %d\n", first_prime);

    /* Part B: Using continue to print only odd numbers from 1 to 15 */
    printf("Odd numbers between 1 and 15: ");
    for (int i = 1; i <= 15; i++) {
        if (i % 2 == 0) {
            continue; /* Skip even numbers directly to i++ */
        }
        printf("%d ", i);
    }
    printf("\n");

    return 0;
}
▶
Output:
First prime >= 50 is: 53
Odd numbers between 1 and 15: 1 3 5 7 9 11 13 15
