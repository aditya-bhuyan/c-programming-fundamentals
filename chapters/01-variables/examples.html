Example 1 — Declaring variables of every basic type

Declares and initialises one variable of each basic type and prints it with the correct format specifier: %d, %u, %ld, %llu, %f, %lf and %c (plus %hd, %lld and %Lf for the remaining types). Integer literal suffixes (u, L, LL, ULL) make each literal's type match its variable.

01_declare.c
#include <stdio.h>

int main(void)
{
    /* Integer types: declaration + initialisation in one statement */
    char               letter  = 'C';                      /* %c   */
    short              year    = 2024;                     /* %hd  */
    int                count   = -42;                      /* %d   */
    unsigned int       items   = 4000000000u;              /* %u   */
    long               pop     = 7900000000L;              /* %ld  */
    long long          deficit = -9000000000000000000LL;   /* %lld */
    unsigned long long huge    = 18000000000000000000ULL;  /* %llu */

    /* Floating-point types */
    float              ratio   = 3.14159f;                 /* %f   */
    double             precise = 2.718281828459045;        /* %f or %lf */
    long double        third   = 1.0L / 3.0L;              /* %Lf  */

    printf("char               : %c (code %d)\n", letter, letter);
    printf("short              : %hd\n", year);
    printf("int                : %d\n", count);
    printf("unsigned int       : %u\n", items);
    printf("long               : %ld\n", pop);
    printf("long long          : %lld\n", deficit);
    printf("unsigned long long : %llu\n", huge);
    printf("float              : %f\n", ratio);
    printf("double (%%f)        : %f\n", precise);
    printf("double (%%lf)       : %lf\n", precise);
    printf("long double        : %Lf\n", third);

    return 0;
}
▶
Output:
char : C (code 67)
short : 2024
int : -42
unsigned int : 4000000000
long : 7900000000
long long : -9000000000000000000
unsigned long long : 18000000000000000000
float : 3.141590
double (%f) : 2.718282
double (%lf) : 2.718282
long double : 0.333333
💡
%f and %lf behave identically in printf because a float argument is promoted to double. (In scanf they differ: %f reads a float, %lf reads a double.) The long double value is only printed to 6 digits here; use %.20Lf to see more.

Example 2 — Sizes and limits with sizeof and <limits.h>

Prints the size of every primitive type with sizeof and %zu, shows both forms of sizeof (with a type and with a variable), and prints INT_MAX, INT_MIN and CHAR_MAX.

02_sizeof.c
#include <stdio.h>
#include <limits.h>   /* CHAR_BIT, CHAR_MAX, INT_MAX, INT_MIN */

int main(void)
{
    int counter = 0;

    /* sizeof(type) - parentheses are REQUIRED for a type name */
    printf("sizeof(char)               = %zu\n", sizeof(char));
    printf("sizeof(short)              = %zu\n", sizeof(short));
    printf("sizeof(int)                = %zu\n", sizeof(int));
    printf("sizeof(long)               = %zu\n", sizeof(long));
    printf("sizeof(long long)          = %zu\n", sizeof(long long));
    printf("sizeof(float)              = %zu\n", sizeof(float));
    printf("sizeof(double)             = %zu\n", sizeof(double));
    printf("sizeof(long double)        = %zu\n", sizeof(long double));
    printf("sizeof(void *)             = %zu\n", sizeof(void *));
    printf("sizeof(size_t)             = %zu\n", sizeof(size_t));

    /* sizeof variable - parentheses are optional for an expression */
    printf("sizeof counter             = %zu\n", sizeof counter);

    /* Constants from <limits.h> */
    printf("CHAR_BIT                   = %d\n", CHAR_BIT);
    printf("CHAR_MAX                   = %d\n", CHAR_MAX);
    printf("INT_MAX                    = %d\n", INT_MAX);
    printf("INT_MIN                    = %d\n", INT_MIN);

    return 0;
}
▶
Output:
sizeof(char) = 1
sizeof(short) = 2
sizeof(int) = 4
sizeof(long) = 8
sizeof(long long) = 8
sizeof(float) = 4
sizeof(double) = 8
sizeof(long double) = 8
sizeof(void *) = 8
sizeof(size_t) = 8
sizeof counter = 4
CHAR_BIT = 8
CHAR_MAX = 127
INT_MAX = 2147483647
INT_MIN = -2147483648
💡
Your output may differ: long is 4 bytes on Windows, long double is 16 on x86-64 Linux, void * is 4 on 32-bit systems, and CHAR_MAX is 255 where plain char is unsigned. The output shown is from a 64-bit macOS (Apple silicon) system.

Example 3 — Signed overflow and unsigned wrap-around

Contrasts signed overflow, which is undefined behaviour, with unsigned overflow, which is fully defined as modular arithmetic.

03_overflow.c
#include <stdio.h>
#include <limits.h>

int main(void)
{
    /*
     * SIGNED overflow is UNDEFINED BEHAVIOUR in C. The standard places no
     * requirement on what happens; the compiler may assume it never occurs
     * and optimise accordingly. On most machines, compiled without
     * optimisation, the result "wraps" to INT_MIN (two's complement), but
     * you must never rely on this.
     *
     * 'volatile' stops the compiler from folding the addition at compile
     * time, so the demonstration reflects what the hardware does.
     */
    volatile int big = INT_MAX;
    int wrapped = big + 1;               /* undefined behaviour! */
    printf("INT_MAX          = %d\n", INT_MAX);
    printf("INT_MAX + 1      = %d  (undefined behaviour)\n", wrapped);

    /*
     * UNSIGNED overflow is well defined: arithmetic is performed modulo
     * 2^N, where N is the number of bits in the type. UINT_MAX + 1 gives 0
     * and 0 - 1 gives UINT_MAX. This is guaranteed by the standard.
     */
    unsigned int u = UINT_MAX;
    printf("UINT_MAX         = %u\n", u);
    u = u + 1u;
    printf("UINT_MAX + 1     = %u  (wraps to 0)\n", u);
    u = u - 1u;
    printf("0 - 1            = %u  (wraps to UINT_MAX)\n", u);

    /* Same rule for narrower unsigned types: 255 + 1 wraps to 0 */
    unsigned char byte = 255;
    byte = (unsigned char)(byte + 1);
    printf("uchar 255 + 1    = %d\n", byte);

    return 0;
}
▶
Output:
INT_MAX = 2147483647
INT_MAX + 1 = -2147483648 (undefined behaviour)
UINT_MAX = 4294967295
UINT_MAX + 1 = 0 (wraps to 0)
0 - 1 = 4294967295 (wraps to UINT_MAX)
uchar 255 + 1 = 0
💡
The signed result -2147483648 is what a typical two's-complement machine produces without optimisation, but the C standard allows anything. With optimisation, compilers may legitimately assume x + 1 > x is always true for signed x. Unsigned arithmetic is the opposite: the standard guarantees wrap-around modulo 2N.

Example 4 — Using const correctly

Shows the four const pointer combinations. Read declarations right-to-left: const int *p is a pointer to a constant int; int *const p is a constant pointer to an int.

04_const.c
#include <stdio.h>

int main(void)
{
    int a = 10;
    int b = 20;

    /* 1. const int - the VALUE can never change after initialisation */
    const int max_users = 100;
    /* max_users = 200;   <- compile error: assignment of read-only variable */
    printf("max_users              = %d\n", max_users);

    /* 2. Pointer to const int - cannot modify the data THROUGH the pointer,
          but the pointer itself can be re-pointed elsewhere. */
    const int *ptr_to_const = &a;
    /* *ptr_to_const = 99;  <- compile error: read-only location */
    printf("*ptr_to_const          = %d\n", *ptr_to_const);
    ptr_to_const = &b;                       /* OK: pointer is not const */
    printf("*ptr_to_const (after)  = %d\n", *ptr_to_const);

    /* 3. const pointer to int - the pointer can never be re-pointed, but
          the data it points to can be modified. */
    int *const const_ptr = &a;
    *const_ptr = 15;                         /* OK: data is not const */
    /* const_ptr = &b;      <- compile error: assignment of read-only variable */
    printf("a via const_ptr        = %d\n", *const_ptr);

    /* 4. const pointer to const int - neither can change */
    const int *const locked = &a;
    /* *locked = 1;  and  locked = &b;  are BOTH compile errors */
    printf("*locked                = %d\n", *locked);

    /*
     * Reading the declaration: read right-to-left.
     *   const int *p        -> p is a pointer to a const int
     *   int *const p        -> p is a const pointer to an int
     *   const int *const p  -> p is a const pointer to a const int
     *
     * Casting away const and then writing to a genuinely const object
     * (e.g. max_users) is undefined behaviour - don't do it.
     */
    return 0;
}
▶
Output:
max_users = 100
*ptr_to_const = 10
*ptr_to_const (after) = 20
a via const_ptr = 15
*locked = 15
💡
Each commented-out line is a compile-time error (typically "assignment of read-only variable" or "read-only location"). The compiler enforces const, so you get the error before the program ever runs. Try un-commenting them one at a time.

Example 5 — Implicit conversions: narrowing, widening, char arithmetic

Demonstrates truncation when a double is stored in an int, widening of an int to double, the difference between integer and floating-point division, and arithmetic on char values.

05_conversion.c
#include <stdio.h>

int main(void)
{
    /* Narrowing: double -> int TRUNCATES toward zero (no rounding) */
    double pi = 3.99;
    double neg = -3.99;
    int truncated = pi;                 /* 3 */
    int truncated_neg = neg;            /* -3, not -4 */
    printf("double 3.99  -> int : %d\n", truncated);
    printf("double -3.99 -> int : %d\n", truncated_neg);

    /* Widening: int -> double is value-preserving here */
    int seven = 7;
    double widened = seven;             /* 7.0 */
    printf("int 7        -> double : %.1f\n", widened);

    /* Integer division vs. floating-point division */
    printf("7 / 2                  = %d\n", seven / 2);          /* int / int       */
    printf("7 / 2.0                = %.1f\n", seven / 2.0);      /* int promoted    */
    printf("(double)7 / 2          = %.1f\n", (double)seven / 2);/* explicit cast   */

    /* char arithmetic: chars are promoted to int before the '+' */
    printf("'A' + 1                = %d\n", 'A' + 1);            /* int 66          */
    printf("(char)('A' + 1)        = %c\n", (char)('A' + 1));    /* 'B'             */
    printf("'a' - 'A'              = %d\n", 'a' - 'A');          /* 32              */
    printf("'7' - '0'              = %d\n", '7' - '0');          /* digit -> value  */

    return 0;
}
▶
Output:
double 3.99 -> int : 3
double -3.99 -> int : -3
int 7 -> double : 7.0
7 / 2 = 3
7 / 2.0 = 3.5
(double)7 / 2 = 3.5
'A' + 1 = 66
(char)('A' + 1) = B
'a' - 'A' = 32
'7' - '0' = 7
💡
'A' + 1 has type int (66), so printing it with %d gives 66. To get the character 'B' you must cast back to char (or print the int with %c). The output assumes an ASCII character set.

Example 6 — A table of <limits.h> and <float.h> constants

Prints a right-aligned table of the integer and floating-point limits for your platform. Each constant has its own type, so each row uses the matching specifier (%d, %u, %ld, %lu, %lld, %llu, %e).

06_limits.c
#include <stdio.h>
#include <limits.h>
#include <float.h>

int main(void)
{
    puts("=== <limits.h> ===");

    printf("%-12s %26d\n",   "SCHAR_MIN",  SCHAR_MIN);

    printf("%-12s %26d\n",   "CHAR_MAX",   CHAR_MAX);
    printf("%-12s %26d\n",   "SHRT_MIN",   SHRT_MIN);
    printf("%-12s %26d\n",   "SHRT_MAX",   SHRT_MAX);
    printf("%-12s %26d\n",   "USHRT_MAX",  USHRT_MAX);
    printf("%-12s %26d\n",   "INT_MIN",    INT_MIN);
    printf("%-12s %26d\n",   "INT_MAX",    INT_MAX);
    printf("%-12s %26u\n",   "UINT_MAX",   UINT_MAX);
