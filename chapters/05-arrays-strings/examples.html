Example 1 — Array Basics: Sum and Average

Declare and initialise an integer array, iterate over it with a for loop, then compute its sum and average. Uses the sizeof idiom to avoid hard-coding the element count.

01_array_basic.c
#include <stdio.h>

int main(void) {
    int values[] = { 4, 8, 15, 16, 23, 42 };
    int    n     = sizeof(values) / sizeof(values[0]);

    /* Print all elements */
    printf("Array: ");
    for (int i = 0; i < n; i++) {
        printf("%d", values[i]);
        if (i < n - 1) printf(", ");
    }
    printf("\n");

    /* Compute sum */
    long sum = 0;
    for (int i = 0; i < n; i++)
        sum += values[i];

    double avg = (double)sum / n;

    printf("Sum     : %ld\n", sum);
    printf("Count   : %d\n", n);
    printf("Average : %.2f\n", avg);

    return 0;
}
💡
Expected output:
Array: 4, 8, 15, 16, 23, 42
Sum : 108
Count : 6
Average : 18.00

Example 2 — Passing Arrays to Functions: Bubble Sort

Pass an integer array to a function that sorts it in-place using the classic bubble sort algorithm. The function receives a pointer and the array length separately.

02_array_fn.c
#include <stdio.h>

/* Sort arr[0..n-1] in ascending order using bubble sort */
void bubble_sort(int *arr, int n) {
    for (int pass = 0; pass < n - 1; pass++) {
        int swapped = 0;
        for (int i = 0; i < n - pass - 1; i++) {
            if (arr[i] > arr[i + 1]) {
                int tmp   = arr[i];
                arr[i]     = arr[i + 1];
                arr[i + 1] = tmp;
                swapped = 1;
            }
        }
        if (!swapped) break;   /* early exit if already sorted */
    }
}

void print_array(const int *arr, int n) {
    for (int i = 0; i < n; i++)
        printf("%d%s", arr[i], i < n - 1 ? ", " : "\n");
}

int main(void) {
    int data[] = { 64, 34, 25, 12, 22, 11, 90 };
    int    n    = sizeof(data) / sizeof(data[0]);

    printf("Before: ");
    print_array(data, n);

    bubble_sort(data, n);

    printf("After : ");
    print_array(data, n);

    return 0;
}
💡
Expected output:
Before: 64, 34, 25, 12, 22, 11, 90
After : 11, 12, 22, 25, 34, 64, 90
💡
Because the array decays to a pointer, bubble_sort modifies the original array in main — there is no copy. The swapped flag is an O(n) optimisation: if a full pass produces no swaps, the array is already sorted and the outer loop exits early.

Example 3 — 2-D Array: Matrix Trace

Declare a 3×3 integer matrix, print it in grid format, then compute its trace (the sum of the main diagonal elements where row index equals column index).

03_2d_array.c
#include <stdio.h>

#define N 3

int main(void) {
    int m[N][N] = {
        { 1, 2, 3 },
        { 4, 5, 6 },
        { 7, 8, 9 }
    };

    printf("Matrix:\n");
    for (int r = 0; r < N; r++) {
        for (int c = 0; c < N; c++)
            printf("%4d", m[r][c]);
        printf("\n");
    }

    /* Trace = sum of diagonal elements m[0][0] + m[1][1] + m[2][2] */
    int trace = 0;
    for (int i = 0; i < N; i++)
        trace += m[i][i];

    printf("Trace = %d\n", trace);
    return 0;
}
💡
Expected output:
Matrix:
   1   2   3
   4   5   6
   7   8   9
Trace = 15

Example 4 — String Basics: Literals, Arrays, and strlen

Contrast a string literal (read-only) with a mutable char[], iterate character-by-character, and measure length with strlen.

04_strings.c
#include <stdio.h>
#include <string.h>

int main(void) {
    /* String literal — read-only, stored in static memory */
    const char *literal = "Hello, World!";

    /* char array — mutable copy on the stack */
    char greeting[] = "Hello, World!";

    printf("Literal : %s\n", literal);
    printf("Array   : %s\n", greeting);
    printf("Length  : %zu\n", strlen(greeting));  /* 13, '\0' not counted */
    printf("Size    : %zu\n", sizeof(greeting));  /* 14, includes '\0'   */

    /* Iterate character-by-character until null terminator */
    printf("Chars   : ");
    for (int i = 0; greeting[i] != '\0'; i++)
        printf("[%c]", greeting[i]);
    printf("\n");

    /* Mutate — valid for char[], not for string literal */
    greeting[7] = 'C';
    greeting[8] = '!';
    greeting[9] = '\0';
    printf("Modified: %s\n", greeting);

    return 0;
}
💡
Expected output:
Literal : Hello, World!
Array : Hello, World!
Length : 13
Size : 14
Chars : [H][e][l][l][o][,][ ][W][o][r][l][d][!]
Modified: Hello, C!

Example 5 — String Operations: strcpy, strcat, strcmp, strstr

Demonstrate the most common <string.h> functions with correctly sized buffers. Note the buffer sizing comments — they are not optional.

05_string_ops.c
#include <stdio.h>
#include <string.h>

int main(void) {
    /* --- strcpy --- */
    char dst[32];                         /* large enough for the source */
    strcpy(dst, "Hello");
    printf("strcpy  : \"%s\"\n", dst);

    /* --- strcat --- */
    strcat(dst, ", World!");              /* dst has 32 bytes, total is 13 — safe */
    printf("strcat  : \"%s\"\n", dst);

    /* --- strcmp --- */
    const char *a = "apple";
    const char *b = "banana";
    int cmp = strcmp(a, b);
    if      (cmp < 0) printf("strcmp  : \"%s\" < \"%s\"\n", a, b);
    else if (cmp > 0) printf("strcmp  : \"%s\" > \"%s\"\n", a, b);
    else             printf("strcmp  : \"%s\" == \"%s\"\n", a, b);

    printf("strcmp  : \"abc\" vs \"abc\" = %d\n", strcmp("abc", "abc"));  /* 0 */

    /* --- strstr --- */
    const char *haystack = "the quick brown fox";
    const char *found    = strstr(haystack, "brown");
    if (found)
        printf("strstr  : found \"brown\" at offset %td\n", found - haystack);
    else
        printf("strstr  : not found\n");

    return 0;
}
💡
Expected output:
strcpy : "Hello"
strcat : "Hello, World!"
strcmp : "apple" < "banana"
strcmp : "abc" vs "abc" = 0
strstr : found "brown" at offset 10

Example 6 — Safe String Handling: Reversal and Bounded Copy

Two safe string utilities: an in-place string reversal (no library functions) and a safe string copy using snprintf as a bounded alternative to strcpy.

06_string_danger.c
#include <stdio.h>
#include <string.h>

/* Reverse a mutable char[] in-place without library functions */
void str_reverse(char *s) {
    if (!s || *s == '\0') return;
    int left  = 0;
    int right = (int)strlen(s) - 1;
    while (left < right) {
        char tmp   = s[left];
        s[left]   = s[right];
        s[right]  = tmp;
        left++;
        right--;
    }
}

/* Safe copy: snprintf always null-terminates, returns chars written */
void safe_copy(char *dst, int dst_size, const char *src) {
    int written = snprintf(dst, (size_t)dst_size, "%s", src);
    if (written >= dst_size)
        printf("  [WARNING] source was truncated to fit buffer\n");
}

int main(void) {
    /* --- Reversal --- */
    char word[] = "racecar";
    printf("Before reverse: %s\n", word);
    str_reverse(word);
    printf("After  reverse: %s\n", word);

    char word2[] = "hello";
    str_reverse(word2);
    printf("Reversed hello: %s\n", word2);

    /* --- Safe copy (fits) --- */
    char buf[16];
    safe_copy(buf, sizeof(buf), "Safe!");
    printf("safe_copy OK  : \"%s\"\n", buf);

    /* --- Safe copy (truncated — source too long for buffer) --- */
    safe_copy(buf, sizeof(buf), "This string is definitely longer than 16 bytes");
    printf("safe_copy trunc: \"%s\"\n", buf);

    return 0;
}
💡
Expected output:
Before reverse: racecar
After reverse: racecar
Reversed hello: olleh
safe_copy OK : "Safe!"
[WARNING] source was truncated to fit buffer
safe_copy trunc: "This string is "
💡
snprintf(dst, size, "%s", src) is the canonical safe-copy idiom. It always null-terminates the destination and returns the number of characters that would have been written — checking the return value against dst_size reveals truncation.
