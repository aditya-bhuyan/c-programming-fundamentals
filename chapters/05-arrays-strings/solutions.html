Exercise 1 — Array Statistics

ex1_stats.c
#include <stdio.h>

int main(void) {
    int arr[] = { 42, 7, 19, 85, 3, 56, 71, 28, 14, 63 };
    int    n   = sizeof(arr) / sizeof(arr[0]);

    printf("Array   : ");
    for (int i = 0; i < n; i++)
        printf("%d%s", arr[i], i < n - 1 ? ", " : "\n");

    int  min = arr[0], max = arr[0];
    long sum = 0;

    for (int i = 0; i < n; i++) {
        if (arr[i] < min) min = arr[i];
        if (arr[i] > max) max = arr[i];
        sum += arr[i];
    }

    printf("Min     : %d\n", min);
    printf("Max     : %d\n", max);
    printf("Sum     : %ld\n", sum);
    printf("Average : %.2f\n", (double)sum / n);

    return 0;
}
💡
Key Insight: Initialise min and max to arr[0], not to 0 or INT_MAX. Starting from the first element is the most readable and correct approach, and allows the loop to start at index 0 without any special-casing.
📄
Expected output:
Array   : 42, 7, 19, 85, 3, 56, 71, 28, 14, 63
Min     : 3
Max     : 85
Sum     : 388
Average : 38.80

Exercise 2 — String Reverse

ex2_reverse.c
#include <stdio.h>

/* Reverse s in-place; finds length without strlen */
void reverse_str(char *s) {
    /* Step 1: find length without strlen */
    int len = 0;
    while (s[len] != '\0')
        len++;

    /* Step 2: two-pointer swap */
    int left  = 0;
    int right = len - 1;
    while (left < right) {
        char tmp   = s[left];
        s[left]   = s[right];
        s[right]  = tmp;
        left++;
        right--;
    }
}

int main(void) {
    char s1[] = "algorithm";
    printf("Original : %s\n", s1);
    reverse_str(s1);
    printf("Reversed : %s\n\n", s1);

    char s2[] = "racecar";
    printf("Original : %s\n", s2);
    reverse_str(s2);
    printf("Reversed : %s\n", s2);

    return 0;
}
💡
Key Insight: The two-pointer technique — one starting at the front, one at the back, closing inward — is the canonical O(n) in-place reversal. No temporary buffer is needed: a single char tmp is sufficient to swap. An odd-length string's centre element is never touched because the loop exits when left == right.
📄
Expected output:
Original : algorithm
Reversed : mhtirogla

Original : racecar
Reversed : racecar

Exercise 3 — Binary Search

ex3_bsearch.c
#include <stdio.h>

int binary_search(const int *arr, int n, int target) {
    int low  = 0;
    int high = n - 1;

    while (low <= high) {
        int mid = low + (high - low) / 2;  /* avoids (low+high) overflow */

        if      (arr[mid] == target) return mid;
        else if (arr[mid] <  target) low  = mid + 1;
        else                          high = mid - 1;
    }

    return -1;  /* not found */
}

int main(void) {
    int sorted[] = { 2, 5, 8, 12, 16, 23, 38, 56, 72, 91 };
    int n = sizeof(sorted) / sizeof(sorted[0]);

    printf("Sorted  : ");
    for (int i = 0; i < n; i++)
        printf("%d%s", sorted[i], i < n-1 ? ", " : "\n");

    int targets[] = { 23, 38, 99, 2 };
    int tc = sizeof(targets) / sizeof(targets[0]);

    for (int i = 0; i < tc; i++) {
        int idx = binary_search(sorted, n, targets[i]);
        if (idx >= 0)
            printf("Search %-3d -> found at index %d\n", targets[i], idx);
        else
            printf("Search %-3d -> not found (-1)\n", targets[i]);
    }

    return 0;
}
💡
Key Insight: Binary search runs in O(log n) time by halving the search space each iteration. The critical correctness detail is computing the midpoint as low + (high - low) / 2 rather than (low + high) / 2 — the latter overflows to a negative value when both low and high exceed INT_MAX / 2.
📄
Expected output:
Sorted  : 2, 5, 8, 12, 16, 23, 38, 56, 72, 91
Search 23  -> found at index 5
Search 38  -> found at index 6
Search 99  -> not found (-1)
Search 2   -> found at index 0

Exercise 4 — Word Count

ex4_wordcount.c
#include <stdio.h>

int word_count(const char *s) {
    int count   = 0;
    int in_word = 0;   /* 0 = in whitespace, 1 = inside a word */

    for (int i = 0; s[i] != '\0'; i++) {
        if (s[i] != ' ') {
            if (!in_word) {
                count++;
                in_word = 1;
            }
        } else {
            in_word = 0;
        }
    }
    return count;
}

int main(void) {
    const char *tests[] = {
        "Hello World",
        "  spaces  everywhere  ",
        "oneword",
        "",
        "   "
    };
    int nt = sizeof(tests) / sizeof(tests[0]);

    for (int i = 0; i < nt; i++)
        printf("\"%-24s\" -> %d words\n", tests[i], word_count(tests[i]));

    return 0;
}
💡
Key Insight: The in_word state flag elegantly handles multiple consecutive spaces, leading spaces, and trailing spaces — the word count only increments on the transition from whitespace to non-whitespace. This is the same pattern used inside C standard library functions like strtok.
📄
Expected output:
"Hello World"            -> 2 words
"  spaces  everywhere  " -> 2 words
"oneword"                -> 1 word
""                       -> 0 words
"   "                    -> 0 words

Exercise 5 — Matrix Multiply

ex5_matmul.c
#include <stdio.h>

#define N 3

void print_matrix(const char *label, int m[][N]) {
    printf("%s\n", label);
    for (int r = 0; r < N; r++) {
        for (int c = 0; c < N; c++)
            printf("%4d", m[r][c]);
        printf("\n");
    }
}

void mat_multiply(int A[][N], int B[][N], int C[][N]) {
    for (int i = 0; i < N; i++)
        for (int j = 0; j < N; j++) {
            C[i][j] = 0;
            for (int k = 0; k < N; k++)
                C[i][j] += A[i][k] * B[k][j];
        }
}

int main(void) {
    int A[N][N] = {
        { 1, 2, 3 },
        { 4, 5, 6 },
        { 7, 8, 9 }
    };
    int B[N][N] = {
        { 7, 8, 9 },
        { 2, 3, 4 },
        { 1, 5, 6 }
    };
    int C[N][N];

    print_matrix("Matrix A:", A);
    print_matrix("Matrix B:", B);

    mat_multiply(A, B, C);

    print_matrix("A x B = C:", C);

    return 0;
}
💡
Key Insight: The three-loop structure — i iterates rows of A, j iterates columns of B, k accumulates the dot product — is the textbook O(n³) algorithm for square matrix multiplication. Explicitly zeroing C[i][j] = 0 inside the i,j loops (or declaring C with = {0}) is essential, as uninitialized stack memory contains garbage.
📄
Expected output:
Matrix A:
   1   2   3
   4   5   6
   7   8   9
Matrix B:
   7   8   9
   2   3   4
   1   5   6
A x B = C:
  14  29  31
  44  67  82
  74 105 133
