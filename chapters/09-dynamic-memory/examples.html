Example 1 — Dynamic Integer Array with malloc

Allocate an array of N integers at runtime, fill with values, print, and free. Demonstrates the NULL check pattern.

01_malloc.c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int n = 8;

    /* Allocate n ints — memory is UNINITIALISED */
    int *arr = malloc((size_t)n * sizeof(int));

    /* ALWAYS check the return value */
    if (arr == NULL) {
        fprintf(stderr, "malloc: out of memory\n");
        return 1;
    }

    /* Fill with squares */
    for (int i = 0; i < n; ++i)
        arr[i] = i * i;

    /* Print */
    printf("Squares (n=%d): ", n);
    for (int i = 0; i < n; ++i)
        printf("%d ", arr[i]);
    printf("\n");

    /* Release heap memory and null the pointer */
    free(arr);
    arr = NULL;

    return 0;
}
▶
Compile & Run: gcc -std=c17 -Wall -g -o 01_malloc 01_malloc.c && ./01_malloc
Expected output: Squares (n=8): 0 1 4 9 16 25 36 49

Example 2 — Zero-Filled Struct Array with calloc

Allocate a struct array with calloc; all fields start at zero. Then fill and print the data.

02_calloc.c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int    id;
    double score;
} Student;

int main(void) {
    int n = 4;

    /* calloc zeros every byte — safe baseline for structs */
    Student *students = calloc((size_t)n, sizeof(Student));
    if (!students) { perror("calloc"); return 1; }

    /* Prove zero-initialisation before filling */
    printf("Before fill: students[0].id=%d, score=%.1f\n",
           students[0].id, students[0].score);

    /* Fill data */
    for (int i = 0; i < n; ++i) {
        students[i].id    = 1001 + i;
        students[i].score = 60.0 + i * 8.5;
    }

    printf("\nAfter fill:\n");
    for (int i = 0; i < n; ++i)
        printf("  id=%d  score=%.1f\n", students[i].id, students[i].score);

    free(students);
    students = NULL;
    return 0;
}
💡
Use calloc when you need a zeroed baseline — e.g. arrays of structs, flag arrays, or any data where uninitialised garbage could cause subtle bugs before the first write.

Example 3 — Growing Array with realloc

Start with capacity 4; when full, double the capacity using the safe realloc pattern.

03_realloc.c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    size_t capacity = 4;
    size_t count    = 0;
    int   *items   = malloc(capacity * sizeof(int));

    if (!items) { perror("malloc"); return 1; }

    /* Append 10 values; grow dynamically when full */
    for (int v = 10; v <= 100; v += 10) {
        if (count == capacity) {
            capacity *= 2;   /* double the capacity */
            int *tmp = realloc(items, capacity * sizeof(int));
            if (!tmp) {
                fprintf(stderr, "realloc failed\n");
                free(items);  /* original still valid — free it */
                return 1;
            }
            items = tmp;
            printf("  [grew to capacity %zu]\n", capacity);
        }
        items[count++] = v;
    }

    printf("Items (%zu): ", count);
    for (size_t i = 0; i < count; ++i)
        printf("%d ", items[i]);
    printf("\n");

    free(items);
    return 0;
}
▶
Doubling capacity on each grow gives amortised O(1) per append — the same strategy used by C++ std::vector and Python's list.

Example 4 — Dynamic 2-D Array (Array of Pointers)

Allocate a rows×cols matrix as an array of row pointers. Fill each cell with row * cols + col, print the grid, then free all rows and the pointer array.

04_2d_dynamic.c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int rows = 3, cols = 4;

    /* Step 1: allocate array of row pointers */
    int **matrix = malloc((size_t)rows * sizeof(int *));
    if (!matrix) { perror("malloc rows"); return 1; }

    /* Step 2: allocate each row */
    for (int r = 0; r < rows; ++r) {
        matrix[r] = malloc((size_t)cols * sizeof(int));
        if (!matrix[r]) {
            /* free already-allocated rows before exiting */
            for (int k = 0; k < r; ++k) free(matrix[k]);
            free(matrix);
            return 1;
        }
    }

    /* Step 3: fill with row * cols + col */
    for (int r = 0; r < rows; ++r)
        for (int c = 0; c < cols; ++c)
            matrix[r][c] = r * cols + c;

    /* Step 4: print */
    printf("Dynamic %dx%d matrix:\n", rows, cols);
    for (int r = 0; r < rows; ++r) {
        for (int c = 0; c < cols; ++c)
            printf("%4d", matrix[r][c]);
        printf("\n");
    }

    /* Step 5: free each row, then the pointer array */
    for (int r = 0; r < rows; ++r)
        free(matrix[r]);
    free(matrix);

    return 0;
}
💡
Free in reverse allocation order: rows first, then the pointer array. Freeing the pointer array first would lose the row addresses, causing a leak.

Example 5 — Memory Leak: Detection & Fix

A deliberate leak, the Valgrind/ASan commands to detect it, and the corrected version side-by-side.

05_leak.c
#include <stdio.h>
#include <stdlib.h>

/* --------------------------------------------------
 * BUGGY VERSION — deliberately leaks heap memory
 * -------------------------------------------------- */
void compute_buggy(int n) {
    int *buf = malloc((size_t)n * sizeof(int));
    if (!buf) return;
    for (int i = 0; i < n; ++i) buf[i] = i;
    printf("computed %d values\n", n);
    /* BUG: returns without calling free(buf) — LEAK */
}

/* --------------------------------------------------
 * FIXED VERSION
 * -------------------------------------------------- */
void compute_fixed(int n) {
    int *buf = malloc((size_t)n * sizeof(int));
    if (!buf) return;
    for (int i = 0; i < n; ++i) buf[i] = i;
    printf("computed %d values\n", n);
    free(buf);   /* FIX: release memory before returning */
    buf = NULL;
}

int main(void) {
    compute_buggy(100);   /* leaks 400 bytes */
    compute_fixed(100);   /* no leak */
    return 0;
}

/*
 * Detect with Valgrind:
 *   gcc -g -o 05_leak 05_leak.c
 *   valgrind --leak-check=full ./05_leak
 *
 * Detect with AddressSanitizer:
 *   gcc -g -fsanitize=address -o 05_leak 05_leak.c
 *   ./05_leak
 */
▶
Valgrind will report: "LEAK SUMMARY: definitely lost: 400 bytes in 1 blocks" for compute_buggy and a clean summary for compute_fixed.

Example 6 — Dynamic String Builder

Start with a 16-byte buffer and double its capacity whenever appending would overflow — a simplified StringBuilder in C.

06_dynstring.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char   *buf;
    size_t  len;      /* number of characters (excluding NUL) */
    size_t  capacity;  /* allocated bytes */
} DynStr;

int ds_init(DynStr *ds) {
    ds->capacity = 16;
    ds->len      = 0;
    ds->buf      = malloc(ds->capacity);
    if (!ds->buf) return 0;
    ds->buf[0] = '\0';
    return 1;
}

int ds_append(DynStr *ds, const char *s) {
    size_t slen = strlen(s);

    /* Grow until there is room for s plus NUL */
    while (ds->len + slen + 1 > ds->capacity) {
        size_t newcap = ds->capacity * 2;
        char  *tmp   = realloc(ds->buf, newcap);
        if (!tmp) return 0;
        ds->buf      = tmp;
        ds->capacity = newcap;
    }

    memcpy(ds->buf + ds->len, s, slen + 1);  /* copy including NUL */
    ds->len += slen;
    return 1;
}

void ds_free(DynStr *ds) {
    free(ds->buf);
    ds->buf      = NULL;
    ds->len      = 0;
    ds->capacity = 0;
}

int main(void) {
    DynStr sb;
    if (!ds_init(&sb)) return 1;

    ds_append(&sb, "Hello, ");
    printf("[cap=%zu] \"%s\"\n", sb.capacity, sb.buf);

    ds_append(&sb, "dynamic ");
    ds_append(&sb, "strings ");
    ds_append(&sb, "are powerful!");
    printf("[cap=%zu len=%zu] \"%s\"\n", sb.capacity, sb.len, sb.buf);

    ds_free(&sb);
    return 0;
}
💡
Encapsulating allocation logic inside ds_init / ds_append / ds_free creates a clean ownership contract: the caller never touches raw malloc/free directly.
