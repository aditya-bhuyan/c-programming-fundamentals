Example 1 — Writing Text with fprintf

Opening a file in write mode ("w"), writing 5 formatted lines with error checks, and closing the stream.

01_write_text.c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    const char *filename = "output_lines.txt";

    /* Open file for writing; creates file or truncates if existing */
    FILE *fp = fopen(filename, "w");
    if (fp == NULL) {
        perror("Failed to open file for writing");
        return EXIT_FAILURE;
    }

    /* Write 5 formatted lines */
    for (int i = 1; i <= 5; ++i) {
        int written = fprintf(fp, "Line %d: The square of %d is %d\n", i, i, i * i);
        if (written < 0) {
            perror("Error writing to file");
            fclose(fp);
            return EXIT_FAILURE;
        }
    }

    if (fclose(fp) == EOF) {
        perror("Error closing file");
        return EXIT_FAILURE;
    }

    printf("Successfully wrote 5 lines to %s\n", filename);
    return EXIT_SUCCESS;
}
▶
Compile & Run: gcc -std=c17 -Wall -Wextra -o 01_write_text 01_write_text.c && ./01_write_text
Creates output_lines.txt containing the 5 generated lines.

Example 2 — Reading Text Line-by-Line with fgets

Opening the file generated in Example 1 in read mode ("r") and safely printing each line with line numbers.

02_read_text.c
#include <stdio.h>
#include <stdlib.h>

#define BUFFER_SIZE 256

int main(void) {
    const char *filename = "output_lines.txt";

    FILE *fp = fopen(filename, "r");
    if (fp == NULL) {
        perror("Error opening file for reading");
        return EXIT_FAILURE;
    }

    char buffer[BUFFER_SIZE];
    int line_num = 0;

    /* fgets returns NULL when EOF is reached or on read error */
    while (fgets(buffer, sizeof(buffer), fp) != NULL) {
        ++line_num;
        printf("[%d] %s", line_num, buffer);
    }

    if (ferror(fp)) {
        perror("I/O error occurred during fgets");
    }

    fclose(fp);
    return EXIT_SUCCESS;
}
▶
Output: Reads each line and prints [1] Line 1: The square of 1 is 1 through [5] Line 5: The square of 5 is 25.

Example 3 — CSV File Writing and Parsing

Writing a CSV dataset of student records, then parsing the CSV line-by-line and printing an aligned tabular summary.

03_csv.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char   name[32];
    int    age;
    double gpa;
} Student;

void write_csv(const char *filename, const Student *list, size_t count) {
    FILE *fp = fopen(filename, "w");
    if (fp == NULL) {
        perror("Failed to open CSV for writing");
        return;
    }

    /* Header row */
    fprintf(fp, "name,age,gpa\n");

    for (size_t i = 0; i < count; ++i) {
        fprintf(fp, "%s,%d,%.2f\n", list[i].name, list[i].age, list[i].gpa);
    }

    fclose(fp);
}

void read_and_print_csv(const char *filename) {
    FILE *fp = fopen(filename, "r");
    if (fp == NULL) {
        perror("Failed to open CSV for reading");
        return;
    }

    char line[128];

    /* Skip header */
    if (fgets(line, sizeof(line), fp) == NULL) {
        fclose(fp);
        return;
    }

    printf("%-15s %-5s %-5s\n", "Name", "Age", "GPA");
    printf("---------------------------\n");

    while (fgets(line, sizeof(line), fp) != NULL) {
        char name[32];
        int age;
        double gpa;

        /* Parse comma-delimited tokens using sscanf */
        if (sscanf(line, "%31[^,],%d,%lf", name, &age, &gpa) == 3) {
            printf("%-15s %-5d %-5.2f\n", name, age, gpa);
        }
    }

    fclose(fp);
}

int main(void) {
    Student students[] = {
        {"Alice Smith", 20, 3.92},
        {"Bob Johnson", 22, 3.45},
        {"Charlie Brown", 19, 3.78},
        {"Diana Prince", 21, 4.00}
    };
    size_t count = sizeof(students) / sizeof(students[0]);

    write_csv("students.csv", students, count);
    read_and_print_csv("students.csv");

    return EXIT_SUCCESS;
}
💡
The scanset specifier %31[^,] reads up to 31 characters until encountering a comma.

Example 4 — Binary Struct Array I/O (fwrite / fread)

Writing an entire array of structs in one call with fwrite, and reading it back into memory with fread.

04_binary.c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int    id;
    char   name[32];
    double balance;
} Account;

int main(void) {
    const char *bin_file = "accounts.bin";

    Account out_accounts[3] = {
        {1001, "Alice", 1450.50},
        {1002, "Bob",    2300.75},
        {1003, "Charlie", 850.00}
    };

    /* 1. Write binary array */
    FILE *fp_out = fopen(bin_file, "wb");
    if (fp_out == NULL) {
        perror("fopen wb failed");
        return EXIT_FAILURE;
    }

    size_t items_written = fwrite(out_accounts, sizeof(Account), 3, fp_out);
    printf("Wrote %zu accounts (%zu bytes total)\n",
           items_written, items_written * sizeof(Account));
    fclose(fp_out);

    /* 2. Read back into a new buffer */
    Account in_accounts[3];
    FILE *fp_in = fopen(bin_file, "rb");
    if (fp_in == NULL) {
        perror("fopen rb failed");
        return EXIT_FAILURE;
    }

    size_t items_read = fread(in_accounts, sizeof(Account), 3, fp_in);
    printf("Read %zu accounts from binary file:\n", items_read);

    for (size_t i = 0; i < items_read; ++i) {
        printf("  ID: %d | Name: %-10s | Balance: $%.2f\n",
               in_accounts[i].id, in_accounts[i].name, in_accounts[i].balance);
    }

    fclose(fp_in);
    return EXIT_SUCCESS;
}
▶
Binary I/O preserves precision exactly without string parsing overhead, but requires careful attention to alignment and portability.

Example 5 — Random Access and In-Place Update with fseek

Writing 10 integers (100–1000), seeking directly to the 5th integer, modifying it in-place using "r+b" mode, and reading back to verify.

05_fseek.c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    const char *filename = "numbers.dat";

    /* Step 1: Create and write 10 integers: 100, 200, 300, ..., 1000 */
    FILE *fp = fopen(filename, "wb+");
    if (fp == NULL) {
        perror("fopen failed");
        return EXIT_FAILURE;
    }

    for (int i = 1; i <= 10; ++i) {
        int val = i * 100;
        fwrite(&val, sizeof(int), 1, fp);
    }

    /* Step 2: Jump to index 4 (the 5th integer, byte offset 4 * sizeof(int)) */
    long offset = 4L * sizeof(int);
    fseek(fp, offset, SEEK_SET);
    printf("Current file position: byte %ld\n", ftell(fp));

    int original;
    fread(&original, sizeof(int), 1, fp);
    printf("Original 5th integer: %d\n", original);

    /* Step 3: Reposition back to 5th integer and update it */
    fseek(fp, offset, SEEK_SET);
    int updated = 9999;
    fwrite(&updated, sizeof(int), 1, fp);
    printf("Updated 5th integer to %d\n", updated);

    /* Step 4: Rewind and read all 10 integers to verify */
    rewind(fp);
    printf("All elements in file: ");
    int val;
    while (fread(&val, sizeof(int), 1, fp) == 1) {
        printf("%d ", val);
    }
    printf("\n");

    fclose(fp);
    return EXIT_SUCCESS;
}
▶
Output shows: All elements in file: 100 200 300 400 9999 600 700 800 900 1000.

Example 6 — Robust Error Handling & EOF Diagnosis

Handling missing files, printing system errors with perror, and distinguishing between natural EOF and I/O corruption with ferror/feof.

06_error.c
#include <stdio.h>
#include <stdlib.h>
#include <errno.h>
#include <string.h>

void attempt_missing_file(void) {
    printf("--- 1. Attempting to open non-existent file ---\n");
    FILE *fp = fopen("does_not_exist_12345.txt", "r");
    if (fp == NULL) {
        /* perror prints custom prefix + ": " + strerror(errno) */
        perror("fopen error");
        printf("errno value: %d (%s)\n", errno, strerror(errno));
    } else {
        fclose(fp);
    }
}

void demonstrate_eof_vs_error(void) {
    printf("\n--- 2. Reading until EOF with proper status checks ---\n");
    FILE *fp = fopen("temp_test.txt", "w+");
    if (fp == NULL) return;

    fputs("Alpha\nBeta\n", fp);
    rewind(fp);

    char buf[64];
    while (fgets(buf, sizeof(buf), fp) != NULL) {
        printf("Read line: %s", buf);
    }

    /* Why did fgets return NULL? */
    if (ferror(fp)) {
        fprintf(stderr, "Stream error encountered!\n");
        clearerr(fp);
    } else if (feof(fp)) {
        printf("Reached end-of-file normally (feof is true).\n");
    }

    fclose(fp);
    remove("temp_test.txt");
}

int main(void) {
    attempt_missing_file();
    demonstrate_eof_vs_error();
    return EXIT_SUCCESS;
}
▶
Output shows fopen error: No such file or directory followed by the graceful EOF diagnosis.
