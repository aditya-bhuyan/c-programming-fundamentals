Exercise 1 — Line Counter

ex1_solution.c
#include <stdio.h>
#include <stdlib.h>

void create_sample_file(const char *filename) {
    FILE *fp = fopen(filename, "w");
    if (fp == NULL) return;
    fputs("First line of text\n", fp);
    fputs("Second line with some numbers 12345\n", fp);
    fputs("Third line here\n", fp);
    fputs("Fourth and final line\n", fp);
    fclose(fp);
}

int count_file_lines(const char *filename) {
    FILE *fp = fopen(filename, "r");
    if (fp == NULL) {
        perror("Failed to open file");
        return -1;
    }

    int ch;
    int lines = 0;
    int last_ch = '\n';

    while ((ch = fgetc(fp)) != EOF) {
        if (ch == '\n') {
            ++lines;
        }
        last_ch = ch;
    }

    /* Count last line if file doesn't end with a newline */
    if (last_ch != '\n' && lines >= 0) {
        ++lines;
    }

    fclose(fp);
    return lines;
}

int main(void) {
    const char *filename = "sample.txt";
    create_sample_file(filename);

    int total = count_file_lines(filename);
    printf("File: %s\n", filename);
    printf("Total lines: %d\n", total);

    remove(filename);
    return EXIT_SUCCESS;
}
💡
Output:
File: sample.txt
Total lines: 4

Exercise 2 — File Copy (Character by Character)

ex2_solution.c
#include <stdio.h>
#include <stdlib.h>

long copy_file(const char *src_path, const char *dst_path) {
    FILE *src = fopen(src_path, "rb");
    if (src == NULL) {
        perror("Failed to open source file");
        return -1L;
    }

    FILE *dst = fopen(dst_path, "wb");
    if (dst == NULL) {
        perror("Failed to open destination file");
        fclose(src);
        return -1L;
    }

    int ch;
    long bytes_copied = 0;

    while ((ch = fgetc(src)) != EOF) {
        if (fputc(ch, dst) == EOF) {
            perror("Write error during copy");
            fclose(src);
            fclose(dst);
            return -1L;
        }
        ++bytes_copied;
    }

    fclose(src);
    fclose(dst);
    return bytes_copied;
}

int main(void) {
    const char *src_file = "source.txt";
    const char *dst_file = "destination.txt";

    /* Prepare source content */
    FILE *fp = fopen(src_file, "w");
    if (fp) {
        fputs("File I/O in C: Character-by-character copy verified.\n", fp);
        fclose(fp);
    }

    long copied = copy_file(src_file, dst_file);
    if (copied >= 0) {
        printf("Copied %ld bytes from '%s' to '%s' successfully.\n",
               copied, src_file, dst_file);
    }

    remove(src_file);
    remove(dst_file);
    return EXIT_SUCCESS;
}
💡
Using "rb" and "wb" guarantees that all characters (including newlines and binary byte values) are preserved exactly without OS newline modifications.

Exercise 3 — CSV Parser & Average GPA Calculator

ex3_solution.c
#include <stdio.h>
#include <stdlib.h>

void create_grades_csv(const char *filename) {
    FILE *fp = fopen(filename, "w");
    if (fp == NULL) return;
    fprintf(fp, "Name,Major,GPA\n");
    fprintf(fp, "Emma Watson,Computer Science,3.90\n");
    fprintf(fp, "Liam Neeson,Mathematics,3.40\n");
    fprintf(fp, "Sophia Chen,Physics,3.80\n");
    fprintf(fp, "Noah Davis,Engineering,3.70\n");
    fclose(fp);
}

int main(void) {
    const char *filename = "grades.csv";
    create_grades_csv(filename);

    FILE *fp = fopen(filename, "r");
    if (fp == NULL) {
        perror("Error opening CSV");
        return EXIT_FAILURE;
    }

    char line[256];
    int student_count = 0;
    double total_gpa = 0.0;

    /* Skip header line */
    if (fgets(line, sizeof(line), fp) == NULL) {
        fclose(fp);
        return EXIT_FAILURE;
    }

    printf("Student Records:\n");

    while (fgets(line, sizeof(line), fp) != NULL) {
        char name[64];
        char major[64];
        double gpa;

        if (sscanf(line, "%63[^,],%63[^,],%lf", name, major, &gpa) == 3) {
            printf("- %s (%s): GPA %.2f\n", name, major, gpa);
            ++student_count;
            total_gpa += gpa;
        }
    }

    fclose(fp);

    if (student_count > 0) {
        printf("\nTotal students: %d\n", student_count);
        printf("Average GPA: %.2f\n", total_gpa / student_count);
    }

    remove(filename);
    return EXIT_SUCCESS;
}
💡
Reads all rows and prints average GPA: (3.90 + 3.40 + 3.80 + 3.70) / 4 = 3.70.

Exercise 4 — Binary Record Database with Random Access Search

ex4_solution.c
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int    id;
    char   name[32];
    double price;
} Product;

int find_product_by_index(FILE *fp, int index, Product *out) {
    if (fp == NULL || out == NULL || index < 0) return 0;

    long offset = (long)index * sizeof(Product);
    if (fseek(fp, offset, SEEK_SET) != 0) {
        return 0;
    }

    size_t n = fread(out, sizeof(Product), 1, fp);
    return n == 1;
}

int main(void) {
    const char *db_path = "catalog.bin";

    Product catalog[5] = {
        {101, "Gaming Mouse",        49.99},
        {102, "4K Monitor",          299.99},
        {103, "Mechanical Keyboard", 89.99},
        {104, "Noise-Canceling Headset", 149.00},
        {105, "USB-C Hub",           29.50}
    };

    /* Write all records */
    FILE *fp_write = fopen(db_path, "wb");
    if (fp_write == NULL) return EXIT_FAILURE;
    fwrite(catalog, sizeof(Product), 5, fp_write);
    fclose(fp_write);
    printf("Database created with 5 products.\n");

    /* Random-access queries */
    FILE *fp_read = fopen(db_path, "rb");
    if (fp_read == NULL) return EXIT_FAILURE;

    Product p;

    printf("Looking up product at index 2 (offset 2 * sizeof(Product))...\n");
    if (find_product_by_index(fp_read, 2, &p)) {
        printf("Found: ID = %d, Name = %s, Price = $%.2f\n\n", p.id, p.name, p.price);
    }

    printf("Looking up product at index 4 (offset 4 * sizeof(Product))...\n");
    if (find_product_by_index(fp_read, 4, &p)) {
        printf("Found: ID = %d, Name = %s, Price = $%.2f\n", p.id, p.name, p.price);
    }

    fclose(fp_read);
    remove(db_path);
    return EXIT_SUCCESS;
}
💡
Random access allows jumping directly to any $O(1)$ record position in files with fixed-size struct records.

Exercise 5 — Log File Rotator

ex5_solution.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_LOG_SIZE 1024L /* 1 KB limit */

long get_file_size(const char *filename) {
    FILE *fp = fopen(filename, "rb");
    if (fp == NULL) return 0L;
    fseek(fp, 0L, SEEK_END);
    long size = ftell(fp);
    fclose(fp);
    return size;
}

void log_message(const char *filename, const char *msg) {
    long current_size = get_file_size(filename);

    /* Check if appending would exceed MAX_LOG_SIZE */
    if (current_size >= MAX_LOG_SIZE) {
        char backup_name[128];
        snprintf(backup_name, sizeof(backup_name), "%s.bak", filename);

        printf("Log size exceeded %ld bytes (%ld bytes). Rotating '%s' -> '%s'...\n",
               MAX_LOG_SIZE, current_size, filename, backup_name);

        remove(backup_name);            /* Remove existing backup if present */
        if (rename(filename, backup_name) != 0) {
            perror("Rotation rename failed");
            return;
        }
        printf("Rotation complete. Fresh '%s' created.\n", filename);
    }

    FILE *fp = fopen(filename, "a");
    if (fp == NULL) {
        perror("Failed to open log file");
        return;
    }

    fprintf(fp, "[LOG] %s\n", msg);
    fclose(fp);
}

int main(void) {
    const char *log_file = "app.log";
    remove(log_file);
    remove("app.log.bak");

    printf("Writing log entries...\n");

    /* Write enough lines to exceed 1 KB */
    for (int i = 1; i <= 25; ++i) {
        char msg[128];
        snprintf(msg, sizeof(msg), "Event #%02d: System operating within normal parameters", i);

        long size_before = get_file_size(log_file);
        log_message(log_file, msg);

        if (i == 1 || i == 22 || i == 24) {
            printf("Entry %d written (size: %ld bytes)\n", i, get_file_size(log_file));
        }
    }

    remove(log_file);
    remove("app.log.bak");
    return EXIT_SUCCESS;
}
💡
Automated log rotation prevents unconstrained disk consumption in production servers and daemons.
