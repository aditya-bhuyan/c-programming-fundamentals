Exercise 1 — Student Record

ex1_solution.c
#include <stdio.h>

typedef struct {
    char   name[32];
    int    age;
    double gpa;
} Student;

/* Simple insertion sort, descending by GPA */
void sort_by_gpa(Student *s, size_t n) {
    for (size_t i = 1; i < n; i++) {
        Student key = s[i];
        size_t j = i;
        while (j > 0 && s[j - 1].gpa < key.gpa) {
            s[j] = s[j - 1];
            j--;
        }
        s[j] = key;
    }
}

int main(void) {
    Student students[3] = {
        { "Alice", 20, 3.45 },
        { "Bob",   22, 3.90 },
        { "Carol", 21, 3.12 }
    };
    size_t n = sizeof(students) / sizeof(students[0]);

    sort_by_gpa(students, n);

    printf("%-8s %-4s %s\n", "Name", "Age", "GPA");
    for (size_t i = 0; i < n; i++) {
        printf("%-8s %-4d %.2f\n", students[i].name, students[i].age, students[i].gpa);
    }

    return 0;
}
Expected output
Name     Age  GPA
Bob      22   3.90
Alice    20   3.45
Carol    21   3.12
💡
Struct assignment (s[j] = s[j - 1]) copies every member, so sorting whole records is as simple as sorting integers. The comparison key is gpa; < in the while condition gives descending order. In production, qsort with a comparator is the better tool.

Exercise 2 — Point Distance

ex2_solution.c
#include <stdio.h>
#include <math.h>

typedef struct {
    double x;
    double y;
} Point;

double distance(Point a, Point b) {
    double dx = b.x - a.x;
    double dy = b.y - a.y;
    return sqrt(dx * dx + dy * dy);
}

int main(void) {
    Point p1 = {0.0, 0.0};
    Point p2 = {3.0, 4.0};
    Point p3 = {-1.5, 2.0};
    Point p4 = {4.5, -2.0};

    printf("Distance (0,0) to (3,4)       = %.4f\n", distance(p1, p2));
    printf("Distance (-1.5,2) to (4.5,-2) = %.4f\n", distance(p3, p4));

    return 0;
}
Expected output
Distance (0,0) to (3,4)       = 5.0000
Distance (-1.5,2) to (4.5,-2) = 7.2111
💡
Structs of two doubles (16 bytes) are cheap to pass by value. Comparing floating-point results exactly is fragile; here 3-4-5 happens to give exactly 5.0, but print with a precision (%.4f) rather than testing with ==.

Exercise 3 — Linked List Node

ex3_solution.c
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int          value;
    struct Node *next;
} Node;

Node *push_front(Node *head, int value) {
    Node *n = malloc(sizeof *n);
    if (n == NULL) {
        return head;
    }
    n->value = value;
    n->next  = head;
    return n;
}

void print_list(const Node *head) {
    for (const Node *cur = head; cur != NULL; cur = cur->next) {
        printf("%d -> ", cur->value);
    }
    printf("NULL\n");
}

void free_list(Node *head) {
    while (head != NULL) {
        Node *next = head->next;
        free(head);
        head = next;
    }
}

int main(void) {
    Node *head = NULL;

    /* Pushing 50, 40, ..., 10 to the front yields 10 -> 20 -> 30 -> 40 -> 50 */
    for (int i = 5; i >= 1; i--) {
        head = push_front(head, i * 10);
    }

    print_list(head);
    free_list(head);
    head = NULL;
    printf("List freed.\n");

    return 0;
}
Expected output
10 -> 20 -> 30 -> 40 -> 50 -> NULL
List freed.
💡
Each node is allocated with malloc(sizeof *n), which stays correct if the type of n changes. In free_list, the next pointer is saved before free(head); reading head->next after freeing would be use-after-free. Setting head = NULL afterwards avoids a dangling pointer.

Exercise 4 — Tagged Union Calculator

ex4_solution.c
#include <stdio.h>

typedef enum { RESULT_INT, RESULT_DOUBLE } ResultType;

typedef struct {
    ResultType type;
    union {
        int    i;
        double d;
    } value;
} Result;

/* Integer division stays an int when exact; otherwise promotes to double */
Result divide(int a, int b) {
    Result r;
    if (a % b == 0) {
        r.type = RESULT_INT;
        r.value.i = a / b;
    } else {
        r.type = RESULT_DOUBLE;
        r.value.d = (double)a / b;
    }
    return r;
}

Result add(int a, int b) {
    Result r;
    r.type = RESULT_INT;
    r.value.i = a + b;
    return r;
}

void print_result(const char *expr, Result r) {
    printf("%-8s = ", expr);
    if (r.type == RESULT_INT) {
        printf("%d (int)\n", r.value.i);
    } else {
        printf("%.4f (double)\n", r.value.d);
    }
}

int main(void) {
    print_result("7 + 5",  add(7, 5));
    print_result("12 / 4", divide(12, 4));
    print_result("7 / 2",  divide(7, 2));
    print_result("10 / 3", divide(10, 3));

    return 0;
}
Expected output
7 + 5    = 12 (int)
12 / 4   = 3 (int)
7 / 2    = 3.5000 (double)
10 / 3   = 3.3333 (double)
💡
The tag decides which union member is read, so the correct format specifier is always chosen. 12 / 4 divides exactly and stays an int, while 7 / 2 promotes to double.

Exercise 5 — Binary File Record

ex5_solution.c
#include <stdio.h>
#include <string.h>

#define FILENAME "records.bin"

typedef struct {
    int    id;
    char   name[24];
    double balance;
} Record;

int main(void) {
    Record out[3] = {
        { 1, "Alice", 1250.75 },
        { 2, "Bob",    310.00 },
        { 3, "Carol", 9999.99 }
    };
    size_t count = sizeof(out) / sizeof(out[0]);

    /* --- Write --- */
    FILE *fp = fopen(FILENAME, "wb");
    if (fp == NULL) {
        perror("fopen (write)");
        return 1;
    }
    size_t written = fwrite(out, sizeof(Record), count, fp);
    fclose(fp);
    printf("Wrote %zu records (%zu bytes each) to %s\n", written, sizeof(Record), FILENAME);

    /* --- Read back --- */
    Record in[3];
    memset(in, 0, sizeof(in));

    fp = fopen(FILENAME, "rb");
    if (fp == NULL) {
        perror("fopen (read)");
        return 1;
    }
    size_t nread = fread(in, sizeof(Record), count, fp);
    fclose(fp);
    printf("Read  %zu records back\n\n", nread);

    for (size_t i = 0; i < nread; i++) {
        printf("id=%d name=%-6s balance=%.2f\n", in[i].id, in[i].name, in[i].balance);
    }

    return 0;
}
Expected output
Wrote 3 records (40 bytes each) to records.bin
Read  3 records back

id=1 name=Alice  balance=1250.75
id=2 name=Bob    balance=310.00
id=3 name=Carol  balance=9999.99
💡
fwrite/fread return the number of items transferred, which should equal count. The binary file stores the raw bytes of each struct, including padding, so it is only portable between systems with the same layout and endianness.
