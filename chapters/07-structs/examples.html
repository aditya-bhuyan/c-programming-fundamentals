Example 1 — Struct Basics: Point and Rectangle

Define Point and Rectangle structs, nest one inside the other, compute an area and print every member.

01_struct_basic.c
#include <stdio.h>

struct Point {
    int x;
    int y;
};

struct Rectangle {
    struct Point top_left;
    struct Point bottom_right;
};

int main(void) {
    struct Point p = {3, 4};
    struct Rectangle r = { {1, 6}, {7, 2} };

    int width  = r.bottom_right.x - r.top_left.x;
    int height = r.top_left.y - r.bottom_right.y;
    int area   = width * height;

    printf("Point p = (%d, %d)\n", p.x, p.y);
    printf("Rectangle top-left     = (%d, %d)\n", r.top_left.x, r.top_left.y);
    printf("Rectangle bottom-right = (%d, %d)\n", r.bottom_right.x, r.bottom_right.y);
    printf("Width = %d, Height = %d, Area = %d\n", width, height, area);

    printf("sizeof(struct Point)     = %zu\n", sizeof(struct Point));
    printf("sizeof(struct Rectangle) = %zu\n", sizeof(struct Rectangle));

    return 0;
}
Output
Point p = (3, 4)
Rectangle top-left     = (1, 6)
Rectangle bottom-right = (7, 2)
Width = 6, Height = 4, Area = 24
sizeof(struct Point)     = 8
sizeof(struct Rectangle) = 16
▶
Compile & Run: gcc -std=c17 -Wall -Wextra -o 01_struct_basic 01_struct_basic.c && ./01_struct_basic
A Rectangle is exactly two Points (2 × 8 bytes): nested structs are stored inline, with no extra overhead.

Example 2 — Passing Structs to Functions

Compare pass-by-value (a copy) with pass-by-pointer (modifies the original), and return a struct from a function.

02_struct_fn.c
#include <stdio.h>

struct Vec2 {
    double x;
    double y;
};

/* Pass by value: the function works on a COPY */
void scale_by_value(struct Vec2 v, double k) {
    v.x *= k;
    v.y *= k;
    printf("  inside scale_by_value: (%.1f, %.1f)\n", v.x, v.y);
}

/* Pass by pointer: the function modifies the ORIGINAL */
void scale_by_pointer(struct Vec2 *v, double k) {
    v->x *= k;
    v->y *= k;
}

/* Return a struct by value */
struct Vec2 vec_add(struct Vec2 a, struct Vec2 b) {
    struct Vec2 result = { a.x + b.x, a.y + b.y };
    return result;
}

int main(void) {
    struct Vec2 v = {1.5, 2.0};

    printf("Original:         (%.1f, %.1f)\n", v.x, v.y);

    scale_by_value(v, 2.0);
    printf("After by-value:   (%.1f, %.1f)\n", v.x, v.y);

    scale_by_pointer(&v, 2.0);
    printf("After by-pointer: (%.1f, %.1f)\n", v.x, v.y);

    struct Vec2 w = {0.5, -1.0};
    struct Vec2 sum = vec_add(v, w);
    printf("vec_add result:   (%.1f, %.1f)\n", sum.x, sum.y);

    return 0;
}
Output
Original:         (1.5, 2.0)
  inside scale_by_value: (3.0, 4.0)
After by-value:   (1.5, 2.0)
After by-pointer: (3.0, 4.0)
vec_add result:   (3.5, 3.0)
▶
Compile & Run: gcc -std=c17 -Wall -Wextra -o 02_struct_fn 02_struct_fn.c && ./02_struct_fn
The by-value call doubled its copy (visible in the inner print) but left v unchanged. Only the pointer version changed the original.

Example 3 — typedef: Struct, Primitive and Function Pointer

Three kinds of typedef in one program, then used in ordinary declarations.

03_typedef.c
#include <stdio.h>

/* 1. typedef for a struct */
typedef struct {
    char name[32];
    int  age;
} Person;

/* 2. typedef for a primitive type */
typedef unsigned char byte;

/* 3. typedef for a function pointer type */
typedef int (*BinaryOp)(int, int);

int add(int a, int b) { return a + b; }
int mul(int a, int b) { return a * b; }

int apply(BinaryOp op, int a, int b) {
    return op(a, b);
}

int main(void) {
    Person alice = { "Alice", 30 };
    printf("Person: %s, age %d\n", alice.name, alice.age);

    byte flags = 0xA5;
    printf("byte flags = 0x%02X (sizeof = %zu)\n", flags, sizeof(byte));

    BinaryOp ops[] = { add, mul };
    const char *names[] = { "add", "mul" };

    for (int i = 0; i < 2; i++) {
        printf("%s(6, 7) = %d\n", names[i], apply(ops[i], 6, 7));
    }

    return 0;
}
Output
Person: Alice, age 30
byte flags = 0xA5 (sizeof = 1)
add(6, 7) = 13
mul(6, 7) = 42
▶
Compile & Run: gcc -std=c17 -Wall -Wextra -o 03_typedef 03_typedef.c && ./03_typedef
BinaryOp lets an array hold functions of the same signature and lets apply accept any of them.

Example 4 — Nested Structs: Employee with a Hire Date

An Employee contains a Date. Pass a const pointer to print a formatted record.

04_nested.c
#include <stdio.h>

typedef struct {
    int day;
    int month;
    int year;
} Date;

typedef struct {
    char   name[32];
    int    id;
    Date   hired;
    double salary;
} Employee;

static const char *MONTHS[] = {
    "January", "February", "March", "April", "May", "June",
    "July", "August", "September", "October", "November", "December"
};

void print_employee(const Employee *e) {
    printf("ID %d: %s\n", e->id, e->name);
    printf("  Hired:  %02d/%02d/%04d (%d %s %d)\n",
           e->hired.day, e->hired.month, e->hired.year,
           e->hired.day, MONTHS[e->hired.month - 1], e->hired.year);
    printf("  Salary: $%.2f\n", e->salary);
}

int main(void) {
    Employee staff[] = {
        { "Alice Johnson", 101, { 15, 3, 2021 }, 72500.00 },
        { "Bob Martinez",  102, {  1, 9, 2019 }, 64250.50 }
    };

    size_t n = sizeof(staff) / sizeof(staff[0]);
    for (size_t i = 0; i < n; i++) {
        print_employee(&staff[i]);
    }

    return 0;
}
Output
ID 101: Alice Johnson
  Hired:  15/03/2021 (15 March 2021)
  Salary: $72500.00
ID 102: Bob Martinez
  Hired:  01/09/2019 (1 September 2019)
  Salary: $64250.50
▶
Compile & Run: gcc -std=c17 -Wall -Wextra -o 04_nested 04_nested.c && ./04_nested
Note e->hired.day: arrow to reach the outer struct, then dot for the nested one.

Example 5 — Tagged Union: A Safe Variant Type

A DataValue union paired with a ValueType enum tag. Constructors set the tag; printing and access check it.

05_union.c
#include <stdio.h>
#include <string.h>

typedef enum { TYPE_INT, TYPE_DOUBLE, TYPE_STRING } ValueType;

typedef union {
    int    i;
    double d;
    char   s[16];
} DataValue;

typedef struct {
    ValueType type;   /* the tag: says which member is valid */
    DataValue data;
} Variant;

Variant make_int(int v) {
    Variant r;
    r.type = TYPE_INT;
    r.data.i = v;
    return r;
}

Variant make_double(double v) {
    Variant r;
    r.type = TYPE_DOUBLE;
    r.data.d = v;
    return r;
}

Variant make_string(const char *v) {
    Variant r;
    r.type = TYPE_STRING;
    strncpy(r.data.s, v, sizeof(r.data.s) - 1);
    r.data.s[sizeof(r.data.s) - 1] = '\0';
    return r;
}

void print_variant(const Variant *v) {
    switch (v->type) {
        case TYPE_INT:    printf("int:    %d\n",     v->data.i); break;
        case TYPE_DOUBLE: printf("double: %.3f\n",   v->data.d); break;
        case TYPE_STRING: printf("string: \"%s\"\n", v->data.s); break;
    }
}

/* Safe accessor: succeeds only if the tag matches */
int get_int(const Variant *v, int *out) {
    if (v->type != TYPE_INT) {
        return 0;
    }
    *out = v->data.i;
    return 1;
}

int main(void) {
    Variant items[] = { make_int(42), make_double(3.14159), make_string("hello") };

    for (size_t i = 0; i < sizeof(items) / sizeof(items[0]); i++) {
        print_variant(&items[i]);
    }

    int x;
    if (get_int(&items[0], &x)) {
        printf("get_int(items[0]) -> %d\n", x);
    }
    if (!get_int(&items[1], &x)) {
        printf("get_int(items[1]) -> rejected (tag is not TYPE_INT)\n");
    }

    printf("sizeof(DataValue) = %zu\n", sizeof(DataValue));
    printf("sizeof(Variant)   = %zu\n", sizeof(Variant));

    return 0;
}
Output
int:    42
double: 3.142
string: "hello"
get_int(items[0]) -> 42
get_int(items[1]) -> rejected (tag is not TYPE_INT)
sizeof(DataValue) = 16
sizeof(Variant)   = 24
▶
Compile & Run: gcc -std=c17 -Wall -Wextra -o 05_union 05_union.c && ./05_union
The union is 16 bytes (its largest member, the 16-char array). Variant is 24 on a typical 64-bit platform: 4-byte tag + 4 bytes padding + 16-byte union (the union needs 8-byte alignment because of double).

Example 6 — Bit-fields: A Permissions Struct

Three 1-bit fields model read / write / execute. Set, test, clear and toggle each flag.

06_bitfield.c
#include <stdio.h>

typedef struct {
    unsigned int read    : 1;
    unsigned int write   : 1;
    unsigned int execute : 1;
} Permissions;

void print_perms(const char *label, Permissions p) {
    printf("%-16s %c%c%c\n", label,
           p.read    ? 'r' : '-',
           p.write   ? 'w' : '-',
           p.execute ? 'x' : '-');
}

int main(void) {
    Permissions perms = {0};          /* all bits cleared */
    print_perms("Initial:", perms);

    perms.read  = 1;                  /* set */
    perms.write = 1;
    print_perms("After set r, w:", perms);

    if (perms.write) {                /* test */
        printf("Write permission is set\n");
    }

    perms.write = 0;                  /* clear */
    print_perms("After clear w:", perms);

    perms.execute = !perms.execute;   /* toggle */
    print_perms("After toggle x:", perms);

    printf("sizeof(Permissions) = %zu\n", sizeof(Permissions));

    return 0;
}
Output
Initial:         ---
After set r, w:  rw-
Write permission is set
After clear w:   r--
After toggle x:  r-x
sizeof(Permissions) = 4
▶
Compile & Run: gcc -std=c17 -Wall -Wextra -o 06_bitfield 06_bitfield.c && ./06_bitfield
Three bits of data still occupy one 4-byte unsigned int storage unit. Sizes and bit order are compiler-dependent.
