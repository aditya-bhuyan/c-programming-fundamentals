Example 1 — Pointer Basics & Dereferencing

Declaring pointers, taking addresses with &, dereferencing with *, and modifying variables indirectly.

01_basics.c
#include <stdio.h>

int main(void) {
    int count = 10;
    int *ptr = &count; /* ptr stores memory address of count */

    printf("count value:              %d\n", count);
    printf("count address (&count):   %p\n", (void *)&count);
    printf("ptr stored address:       %p\n", (void *)ptr);
    printf("ptr own address (&ptr):    %p\n", (void *)&ptr);
    printf("dereference (*ptr):       %d\n", *ptr);

    /* Modify count through ptr */
    *ptr = 25;
    printf("\nAfter '*ptr = 25':\n");
    printf("count value:              %d\n", count);
    printf("dereference (*ptr):       %d\n", *ptr);

    return 0;
}
▶
Compile & Run: gcc -std=c17 -Wall -o 01_basics 01_basics.c && ./01_basics
Notice that &count and ptr print the exact same hexadecimal address.

Example 2 — Pointer Arithmetic vs Array Indexing

Traversing an array using pointer steps and demonstrating how indexing maps directly to pointer math in compiled assembly.

02_arithmetic.c
#include <stdio.h>
#include <stddef.h>

int main(void) {
    int numbers[] = {11, 22, 33, 44, 55};
    size_t len = sizeof(numbers) / sizeof(numbers[0]);

    printf("--- 1. Traditional Index Loop ---\n");
    for (size_t i = 0; i < len; ++i) {
        printf("numbers[%zu] = %d (address: %p)\n",
               i, numbers[i], (void *)&numbers[i]);
    }

    printf("\n--- 2. Pointer Arithmetic Loop ---\n");
    /* ptr++ advances by sizeof(int) = 4 bytes each step */
    for (int *p = numbers; p < numbers + len; ++p) {
        ptrdiff_t idx = p - numbers; /* element offset */
        printf("offset %td: value = %d (address: %p)\n",
               idx, *p, (void *)p);
    }

    /*
     * Compiler equivalence:
     * In x86-64 assembly, numbers[i] and *(numbers + i) compile to:
     *   mov eax, DWORD PTR [rbp - 32 + rdi*4]
     * In both cases, the index register (rdi) is scaled by 4 bytes.
     */

    return 0;
}
▶
Output demonstrates each address incrementing by 4 bytes (e.g. ...00, ...04, ...08, ...0C, ...10).

Example 3 — Pass-by-Reference with Pointers (Swap)

Contrasting broken value-copy swap with pointer-based swap that modifies the caller's variables.

03_swap.c
#include <stdio.h>

/* Broken swap: receives local copies, caller's variables remain untouched */
void swap_by_value(int a, int b) {
    int temp = a;
    a = b;
    b = temp;
}

/* Correct swap: receives memory addresses and dereferences to swap in-place */
void swap_by_pointer(int *a, int *b) {
    if (a == NULL || b == NULL) return; /* Defensive null check */
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main(void) {
    int x = 100, y = 200;

    printf("Initial:        x = %d, y = %d\n", x, y);

    swap_by_value(x, y);
    printf("After value:    x = %d, y = %d (no change!)\n", x, y);

    swap_by_pointer(&x, &y);
    printf("After pointer:  x = %d, y = %d (swapped!)\n", x, y);

    return 0;
}
▶
Because C is strictly pass-by-value, passing addresses (&x, &y) is the canonical way to simulate pass-by-reference.

Example 4 — The Four const Pointer Variations

Detailed compile-time demonstration of mutable vs immutable pointer addresses and values.

04_const_ptr.c
#include <stdio.h>

int main(void) {
    int val1 = 10;
    int val2 = 20;

    /* 1. Non-const pointer to non-const int */
    int *p1 = &val1;
    *p1 = 11;        /* OK: can modify target */
    printf("p1: *p1 = %d\n", *p1);
    p1 = &val2;       /* OK: can reassign pointer */
    printf("p1: *p1 = %d\n", *p1);

    /* 2. Pointer to const int (pointer to constant data) */
    const int *p2 = &val1;
    p2 = &val2;       /* OK: pointer itself can change */
    printf("p2: *p2 = %d\n", *p2);
    /* *p2 = 12; */   /* COMPILER ERROR: assignment of read-only location */

    /* 3. Const pointer to non-const int (constant pointer address) */
    int * const p3 = &val1;
    *p3 = 13;        /* OK: data can be modified */
    printf("p3: *p3 = %d\n", *p3);
    /* p3 = &val2; */ /* COMPILER ERROR: assignment of read-only variable */

    /* 4. Const pointer to const int (everything immutable) */
    const int * const p4 = &val1;
    printf("p4: *p4 = %d\n", *p4);
    /* *p4 = 14;   */ /* COMPILER ERROR */
    /* p4 = &val2; */ /* COMPILER ERROR */

    return 0;
}
💡
Rule: read from right to left starting at the variable identifier: const int * const p is "p is a const pointer to an int that is const".

Example 5 — Modifying Caller Pointers via Double Pointers

Using int ** so a function can redirect where a pointer variable in the calling function points.

05_double_ptr.c
#include <stdio.h>

static int primary_buffer[3] = {10, 20, 30};
static int backup_buffer[3]  = {90, 80, 70};

/* Fails: modifies local copy of pointer */
void switch_buffer_failed(int *buf) {
    buf = backup_buffer;
}

/* Succeeds: receives address of caller's pointer (*buf_ptr) */
void switch_buffer_success(int **buf_ptr) {
    if (buf_ptr != NULL) {
        *buf_ptr = backup_buffer; /* redirect caller's pointer */
    }
}

int main(void) {
    int *active = primary_buffer;
    printf("Initial active[0]: %d\n", active[0]); /* 10 */

    switch_buffer_failed(active);
    printf("After failed switch: %d\n", active[0]); /* still 10 */

    switch_buffer_success(&active);
    printf("After double pointer switch: %d\n", active[0]); /* now 90 */

    return 0;
}
▶
This same Type ** pattern is used extensively in dynamic data structures (e.g. linked lists, tree roots, dynamic array resizing).

Example 6 — Function Pointer Dispatch Table

Building an extensible mathematical expression evaluator table using function pointers.

06_fnptr_table.c
#include <stdio.h>
#include <stdlib.h>

/* Typedef for a binary arithmetic operation */
typedef double (*calc_op)(double, double);

double op_add(double a, double b) { return a + b; }
double op_sub(double a, double b) { return a - b; }
double op_mul(double a, double b) { return a * b; }
double op_div(double a, double b) {
    if (b == 0.0) {
        printf("Error: Division by zero!\n");
        return 0.0;
    }
    return a / b;
}

typedef struct {
    char    symbol;
    calc_op func;
} DispatchEntry;

static const DispatchEntry table[] = {
    {'+', op_add},
    {'-', op_sub},
    {'*', op_mul},
    {'/', op_div}
};

double evaluate(char op, double a, double b) {
    size_t count = sizeof(table) / sizeof(table[0]);
    for (size_t i = 0; i < count; ++i) {
        if (table[i].symbol == op) {
            return table[i].func(a, b); /* call via function pointer */
        }
    }
    printf("Unknown operator '%c'\n", op);
    return 0.0;
}

int main(void) {
    printf("12 + 4 = %.2f\n", evaluate('+', 12, 4));
    printf("12 - 4 = %.2f\n", evaluate('-', 12, 4));
    printf("12 * 4 = %.2f\n", evaluate('*', 12, 4));
    printf("12 / 4 = %.2f\n", evaluate('/', 12, 4));
    return 0;
}
▶
Dispatch tables eliminate long switch statements and allow adding new operations simply by registering new table rows without altering existing logic.
