Exercise 1 — Pointer Basics & Indirect Modification

ex1_solution.c
#include <stdio.h>

int main(void) {
    int    i_val = 42;
    double d_val = 3.14159;
    char   c_val = 'Z';

    int    *p_i = &i_val;
    double *p_d = &d_val;
    char   *p_c = &c_val;

    printf("Initial:\n");
    printf("int val:    %d at address %p\n",    *p_i, (void *)p_i);
    printf("double val: %f at address %p\n", *p_d, (void *)p_d);
    printf("char val:   %c at address %p\n",    *p_c, (void *)p_c);

    /* Modify via pointers */
    *p_i = 100;
    *p_d = 2.71828;
    *p_c = 'A';

    printf("\nAfter modification via pointers:\n");
    printf("int val:    %d\n", i_val);
    printf("double val: %f\n", d_val);
    printf("char val:   %c\n", c_val);

    return 0;
}
💡
Writing *p_i = 100; writes directly to the memory address of i_val, so reading i_val subsequently produces the newly updated value.

Exercise 2 — Array Sum via Pointer Arithmetic

ex2_solution.c
#include <stdio.h>
#include <stddef.h>

long long sum_array(const int *arr, size_t n) {
    long long total = 0;
    const int *end = arr + n; /* pointer to one-past-the-last element */

    while (arr < end) {
        total += *arr;        /* dereference current element */
        arr++;                 /* advance pointer by sizeof(int) */
    }

    return total;
}

int main(void) {
    int data[] = {10, 25, 30, 45, 50, 60};
    size_t count = sizeof(data) / sizeof(data[0]);

    long long result = sum_array(data, count);
    printf("Array: [10, 25, 30, 45, 50, 60]\n");
    printf("Sum = %lld\n", result);

    return 0;
}
💡
Because arr is declared as const int *arr, the function promises not to modify any array elements while summing them.

Exercise 3 — Generic and Typed Pointer Swaps

ex3_solution.c
#include <stdio.h>
#include <stddef.h>

void swap_int(int *a, int *b) {
    if (a && b) {
        int tmp = *a;
        *a = *b;
        *b = tmp;
    }
}

void swap_double(double *a, double *b) {
    if (a && b) {
        double tmp = *a;
        *a = *b;
        *b = tmp;
    }
}

void swap_generic(void *a, void *b, size_t size) {
    if (!a || !b || a == b) return;
    unsigned char *p1 = (unsigned char *)a;
    unsigned char *p2 = (unsigned char *)b;

    for (size_t i = 0; i < size; ++i) {
        unsigned char temp = p1[i];
        p1[i] = p2[i];
        p2[i] = temp;
    }
}

int main(void) {
    int a = 10, b = 20;
    swap_int(&a, &b);
    printf("int swap:    a = 20, b = 10 (actual: a = %d, b = %d)\n", a, b);

    double x = 1.50, y = 9.75;
    swap_double(&x, &y);
    printf("double swap: x = %.2f, y = %.2f\n", x, y);

    long p = 12345, q = 67890;
    swap_generic(&p, &q, sizeof(long));
    printf("generic:     p = %ld, q = %ld\n", p, q);

    return 0;
}
💡
In C, unsigned char * is explicitly allowed to alias and inspect any memory block byte-by-byte without breaking strict aliasing rules.

Exercise 4 — Reverse Array In-Place (Two-Pointer Technique)

ex4_solution.c
#include <stdio.h>
#include <stddef.h>

void reverse_array(int *arr, size_t n) {
    if (arr == NULL || n <= 1) return;

    int *left = arr;
    int *right = arr + n - 1;

    while (left < right) {
        int temp = *left;
        *left = *right;
        *right = temp;

        ++left;
        --right;
    }
}

int main(void) {
    int arr[] = {1, 2, 3, 4, 5, 6, 7};
    size_t len = sizeof(arr) / sizeof(arr[0]);

    printf("Original: ");
    for (size_t i = 0; i < len; ++i) printf("%d ", arr[i]);
    printf("\n");

    reverse_array(arr, len);

    printf("Reversed: ");
    for (size_t i = 0; i < len; ++i) printf("%d ", arr[i]);
    printf("\n");

    return 0;
}
💡
The two-pointer technique achieves $O(n)$ time complexity with $O(1)$ extra space, reversing elements completely in-place.

Exercise 5 — Bounded String Copy Implementation

ex5_solution.c
#include <stdio.h>
#include <stddef.h>

char *my_strncpy(char *dst, const char *src, size_t n) {
    if (dst == NULL || n == 0) return dst;
    if (src == NULL) {
        *dst = '\0';
        return dst;
    }

    char *start = dst;
    size_t copied = 0;

    /* Copy characters up to n - 1 */
    while (*src != '\0' && copied + 1 < n) {
        *dst++ = *src++;
        ++copied;
    }

    /* Guarantee null termination */
    *dst = '\0';

    return start;
}

int main(void) {
    const char *source = "C Programming Fundamentals";
    char buf1[30];
    char buf2[14];

    my_strncpy(buf1, source, sizeof(buf1));
    my_strncpy(buf2, source, sizeof(buf2));

    printf("Source:      \"%s\"\n", source);
    printf("Copied (30): \"%s\"\n", buf1);
    printf("Copied (14): \"%s\"\n", buf2);

    return 0;
}
💡
Unlike the standard C library strncpy (which does not null-terminate if the source length exceeds n), this safe implementation guarantees null-termination without buffer overflows.
