Exercise 1 — Power Function

ex1_power.c
#include <stdio.h>

double power(double base, int exp) {
    if (exp < 0) {
        /* base^-n == 1 / base^n */
        return 1.0 / power(base, -exp);
    }

    double result = 1.0;
    for (int i = 0; i < exp; i++) {
        result *= base;
    }
    return result;
}

int main(void) {
    printf("2 ^ 10  = %g\n", power(2.0, 10));
    printf("2 ^ -3  = %g\n", power(2.0, -3));
    printf("5 ^ 0   = %g\n", power(5.0, 0));
    printf("(-3) ^ 3 = %g\n", power(-3.0, 3));
    printf("10 ^ -2 = %g\n", power(10.0, -2));
    printf("1.5 ^ 2 = %g\n", power(1.5, 2));
    return 0;
}
▶
Compile: gcc -std=c17 -Wall -Wextra -o ex1_power ex1_power.c  |  Run: ./ex1_power
Output:2 ^ 10 = 1024 2 ^ -3 = 0.125 5 ^ 0 = 1 (-3) ^ 3 = -27 10 ^ -2 = 0.01 1.5 ^ 2 = 2.25
💡
Key Insight: A negative exponent is handled by reducing it to the positive case: x-n = 1 / xn. The recursive call power(base, -exp) is made with a positive exponent, so it reaches the loop and cannot recurse again. power(x, 0) returns 1.0 because the loop body never executes.

Exercise 2 — String Length

ex2_strlen.c
#include <stdio.h>

size_t my_strlen(const char *str) {
    size_t length = 0;
    while (str[length] != '\0') {
        length++;
    }
    return length;
}

int main(void) {
    printf("\"Hello\"         -> %zu\n", my_strlen("Hello"));
    printf("\"\"              -> %zu\n", my_strlen(""));
    printf("\"C programming\" -> %zu\n", my_strlen("C programming"));
    printf("\"a b\"           -> %zu\n", my_strlen("a b"));
    return 0;
}
▶
Compile: gcc -std=c17 -Wall -Wextra -o ex2_strlen ex2_strlen.c  |  Run: ./ex2_strlen
Output:"Hello" -> 5 "" -> 0 "C programming" -> 13 "a b" -> 3
💡
Key Insight: A C string has no stored length. The only way to find it is to scan until the null terminator '\0'. This is why strlen is O(n) and why calling it repeatedly in a loop condition is wasteful. The parameter is const char * because the function only reads the string.

Exercise 3 — Recursive Sum of Digits

ex3_digitsum.c
#include <stdio.h>

int sum_digits(int n) {
    if (n < 0) {
        n = -n;                             /* ignore the sign */
    }
    if (n < 10) {
        return n;                           /* base case: a single digit */
    }
    return (n % 10) + sum_digits(n / 10);   /* last digit + sum of the rest */
}

int main(void) {
    printf("sum_digits(12345) = %d\n", sum_digits(12345));
    printf("sum_digits(0)     = %d\n", sum_digits(0));
    printf("sum_digits(9)     = %d\n", sum_digits(9));
    printf("sum_digits(-987)  = %d\n", sum_digits(-987));
    printf("sum_digits(1000)  = %d\n", sum_digits(1000));
    return 0;
}
▶
Compile: gcc -std=c17 -Wall -Wextra -o ex3_digitsum ex3_digitsum.c  |  Run: ./ex3_digitsum
Output:sum_digits(12345) = 15 sum_digits(0) = 0 sum_digits(9) = 9 sum_digits(-987) = 24 sum_digits(1000) = 1
💡
Key Insight: Each call peels off the last digit with n % 10 and recurses on n / 10, which is strictly smaller, so the base case (n < 10) is always reached. sum_digits(1000) shows that zero digits are handled correctly: 0 + 0 + 0 + 1. -n is safe here except for INT_MIN, which cannot be negated in an int.

Exercise 4 — Function Pointer Calculator

ex4_calculator.c
#include <stdio.h>

typedef double (*binary_op)(double, double);

double add(double a, double b)      { return a + b; }
double subtract(double a, double b) { return a - b; }
double multiply(double a, double b) { return a * b; }
double divide(double a, double b)   { return a / b; }

int main(void) {
    binary_op ops[]     = { add, subtract, multiply, divide };
    const char symbol[] = { '+', '-', '*', '/' };
    int num_ops = (int)(sizeof ops / sizeof ops[0]);

    /* Pre-set "user input": operation index, left operand, right operand */
    int    choices[] = { 0, 1, 2, 3, 3, 7 };
    double lefts[]   = { 12.0, 12.0, 12.0, 12.0, 5.0, 1.0 };
    double rights[]  = {  4.0,  4.0,  4.0,  4.0, 0.0, 1.0 };
    int num_tests = (int)(sizeof choices / sizeof choices[0]);

    for (int i = 0; i < num_tests; i++) {
        int c = choices[i];
        if (c < 0 || c >= num_ops) {
            printf("[ERROR] Invalid operation index %d\n", c);
        } else if (symbol[c] == '/' && rights[i] == 0.0) {
            printf("[ERROR] %.2f / %.2f: division by zero\n", lefts[i], rights[i]);
        } else {
            printf("%.2f %c %.2f = %.2f\n",
                   lefts[i], symbol[c], rights[i], ops[c](lefts[i], rights[i]));
        }
    }
    return 0;
}
▶
Compile: gcc -std=c17 -Wall -Wextra -o ex4_calculator ex4_calculator.c  |  Run: ./ex4_calculator
Output:12.00 + 4.00 = 16.00 12.00 - 4.00 = 8.00 12.00 * 4.00 = 48.00 12.00 / 4.00 = 3.00 [ERROR] 5.00 / 0.00: division by zero [ERROR] Invalid operation index 7
💡
Key Insight: The dispatch table replaces a switch with a single indexed call, ops[c](lefts[i], rights[i]). Adding a new operation means writing one function and adding one entry to each array. The bounds check c < 0 || c >= num_ops must come before indexing the table, and division by zero is rejected before the function is called.

Exercise 5 — Tower of Hanoi

ex5_hanoi.c
#include <stdio.h>

/* Moves n disks from `from` to `to`, using `via` as the spare peg.
   Returns the number of moves made. */
unsigned long hanoi(int n, char from, char to, char via) {
    if (n == 0) {
        return 0;                                        /* base case: nothing to move */
    }

    unsigned long moves = hanoi(n - 1, from, via, to);   /* 1. clear the top n-1 disks */
    printf("Move disk %d: %c -> %c\n", n, from, to);     /* 2. move the largest disk   */
    moves++;
    moves += hanoi(n - 1, via, to, from);                /* 3. stack n-1 disks on top  */

    return moves;
}

int main(void) {
    int n = 3;
    printf("Tower of Hanoi with %d disks:\n", n);
    unsigned long total = hanoi(n, 'A', 'C', 'B');
    printf("Total moves: %lu (formula 2^%d - 1 = %lu)\n", total, n, (1UL << n) - 1);
    return 0;
}
▶
Compile: gcc -std=c17 -Wall -Wextra -o ex5_hanoi ex5_hanoi.c  |  Run: ./ex5_hanoi
Output:Tower of Hanoi with 3 disks: Move disk 1: A -> C Move disk 2: A -> B Move disk 1: C -> B Move disk 3: A -> C Move disk 1: B -> A Move disk 2: B -> C Move disk 1: A -> C Total moves: 7 (formula 2^3 - 1 = 7)
💡
Key Insight: Trust the recursion: if you can move n-1 disks, then moving n disks is “move n-1 out of the way, move the biggest disk, move n-1 back on top”. The roles of the pegs swap with each call (via and to exchange places). The move count satisfies T(n) = 2T(n-1) + 1, which solves to 2n - 1, so 20 disks already need over a million moves.
