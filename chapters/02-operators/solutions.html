Exercise 1 — Arithmetic Calculator

ex1_solution.c
#include <stdio.h>

int main(void)
{
    int a = 17;
    int b = 5;

    printf("a = %d, b = %d\n", a, b);
    printf("a + b = %d\n", a + b);
    printf("a - b = %d\n", a - b);
    printf("a * b = %d\n", a * b);
    printf("a / b = %d\n", a / b);          /* integer division */
    printf("a %% b = %d\n", a % b);         /* remainder */
    printf("a / b (real) = %.2f\n", (double)a / b);

    return 0;
}
▶
Expected output:
a = 17, b = 5
a + b = 22
a - b = 12
a * b = 85
a / b = 3
a % b = 2
a / b (real) = 3.40
💡
17 / 5 is 3 because both operands are int; 17 % 5 is the leftover 2. The cast in (double)a / b forces real division, giving 3.40.

Exercise 2 — Bit Flags

ex2_solution.c
#include <stdio.h>

static void print_bin(unsigned char value)
{
    for (int bit = 7; bit >= 0; bit--) {
        putchar(((value >> bit) & 1u) ? '1' : '0');
    }
}

int main(void)
{
    unsigned char flags = 0;

    flags |= (1u << 0);     /* set bit 0 */
    flags |= (1u << 2);     /* set bit 2 */
    flags |= (1u << 4);     /* set bit 4 */

    printf("After setting bits 0, 2, 4: ");
    print_bin(flags);
    printf(" (%u)\n", (unsigned)flags);

    flags &= ~(1u << 2);    /* clear bit 2 */

    printf("After clearing bit 2:       ");
    print_bin(flags);
    printf(" (%u)\n", (unsigned)flags);

    return 0;
}
▶
Expected output:
After setting bits 0, 2, 4: 00010101 (21)
After clearing bit 2: 00010001 (17)
💡
Bits 0, 2, 4 give 1 + 4 + 16 = 21. Clearing bit 2 with &= ~(1u << 2) subtracts 4, leaving 17. The mask ~(1u << 2) has every bit set except bit 2, so the AND clears only that bit.

Exercise 3 — Short-Circuit Demo

ex3_solution.c
#include <stdio.h>

static int first(void)
{
    printf("first() called\n");
    return 0;                   /* false */
}

static int second(void)
{
    printf("second() called\n");
    return 1;                   /* true */
}

int main(void)
{
    int r;

    printf("--- first() && second() ---\n");
    r = first() && second();    /* first() is false -> second() skipped */
    printf("result = %d\n\n", r);

    printf("--- second() && first() ---\n");
    r = second() && first();    /* second() is true -> first() must run */
    printf("result = %d\n\n", r);

    printf("--- first() || second() ---\n");
    r = first() || second();    /* first() is false -> second() must run */
    printf("result = %d\n\n", r);

    printf("--- second() || first() ---\n");
    r = second() || first();    /* second() is true -> first() skipped */
    printf("result = %d\n", r);

    return 0;
}
▶
Expected output:
--- first() && second() ---
first() called
result = 0
 
--- second() && first() ---
second() called
first() called
result = 0
 
--- first() || second() ---
first() called
second() called
result = 1
 
--- second() || first() ---
second() called
result = 1
💡
In first() && second() the left side is false so second() never runs; in second() || first() the left side is true so first() never runs. In the other two expressions both functions are called.

Exercise 4 — Swap Without Temp

ex4_solution.c
#include <stdio.h>

int main(void)
{
    int a = 12;
    int b = 25;

    printf("Before: a = %d, b = %d\n", a, b);

    a = a ^ b;      /* a now holds a XOR b          */
    b = a ^ b;      /* (a ^ b) ^ b  ->  original a  */
    a = a ^ b;      /* (a ^ b) ^ a  ->  original b  */

    printf("After:  a = %d, b = %d\n", a, b);

    return 0;
}
▶
Expected output:
Before: a = 12, b = 25
After: a = 25, b = 12
💡
With a = 12 (01100) and b = 25 (11001): step 1 makes a = 10101; step 2 gives b = 10101 ^ 11001 = 01100 (12); step 3 gives a = 10101 ^ 01100 = 11001 (25). Caveat: this trick fails if both operands are the same object (it zeroes it), and a temporary is usually clearer and just as fast.

Exercise 5 — Precedence Puzzle

ex5_solution.c
#include <stdio.h>

int main(void)
{
    int a = 5, b = 3, c = 2;

    /* 1: * before +        -> 5 + (3 * 2) */
    printf("1) a + b * c          = %d\n", a + b * c);

    /* 2: - is left-assoc   -> (5 - 3) - 2 */
    printf("2) a - b - c          = %d\n", a - b - c);

    /* 3: && before ||      -> ((a > b) && (b > c)) || (c > a) */
    printf("3) a > b && b > c || c > a = %d\n", a > b && b > c || c > a);

    /* 4: == before &       -> a & (b == c) */
    printf("4) a & b == c         = %d\n", a & b == c);

    /* 5: + before <<       -> a << (1 + c) */
    printf("5) a << 1 + c         = %d\n", a << 1 + c);

    return 0;
}
▶
Expected output:
1) a + b * c = 11
2) a - b - c = 0
3) a > b && b > c || c > a = 1
4) a & b == c = 0
5) a << 1 + c = 40
💡
With a = 5, b = 3, c = 2 the groupings are:
1) 5 + (3 * 2) = 11
2) (5 - 3) - 2 = 0
3) ((5 > 3) && (3 > 2)) || (2 > 5) = 1
4) 5 & (3 == 2) = 5 & 0 = 0
5) 5 << (1 + 2) = 40
The compiler warns about lines 3 and 4 (and 5) — adding the parentheses shown above silences the warnings and documents the intent.
