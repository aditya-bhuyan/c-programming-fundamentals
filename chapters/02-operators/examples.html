Example 1 — Arithmetic operators

All five arithmetic operators, the integer-division trap, and how % behaves with negative operands (C99+).

01_arithmetic.c
#include <stdio.h>

int main(void)
{
    int a = 17;
    int b = 5;

    printf("a = %d, b = %d\n\n", a, b);

    /* The five arithmetic operators */
    printf("a + b = %d\n", a + b);
    printf("a - b = %d\n", a - b);
    printf("a * b = %d\n", a * b);
    printf("a / b = %d   <- integer division: fraction discarded\n", a / b);
    printf("a %% b = %d    <- remainder (note %%%% to print a literal %%)\n", a % b);

    /* The integer-division trap */
    printf("\n--- Integer division trap ---\n");
    int x = 7, y = 2;
    double wrong = x / y;           /* int / int happens FIRST -> 3, then converted */
    double right = (double)x / y;   /* cast one operand -> floating-point division  */
    printf("7 / 2          = %d\n", 7 / 2);
    printf("double wrong   = %.2f\n", wrong);
    printf("double right   = %.2f\n", right);

    /* Modulo with negative operands (C99 and later: truncation toward zero) */
    printf("\n--- Modulo with negatives (C99+) ---\n");
    printf(" 7 / 2 = %2d   7 %% 2  = %2d\n",  7 / 2,  7 % 2);
    printf("-7 / 2 = %2d  -7 %% 2  = %2d\n", -7 / 2, -7 % 2);
    printf(" 7 / -2 = %2d   7 %% -2 = %2d\n",  7 / -2,  7 % -2);
    printf("-7 / -2 = %2d  -7 %% -2 = %2d\n", -7 / -2, -7 % -2);

    /* Identity that always holds: (a / b) * b + a % b == a */
    printf("\nIdentity check: (-7 / 2) * 2 + (-7 %% 2) = %d\n", (-7 / 2) * 2 + (-7 % 2));

    return 0;
}
▶
Output:
a = 17, b = 5
 
a + b = 22
a - b = 12
a * b = 85
a / b = 3 <- integer division: fraction discarded
a % b = 2 <- remainder (note %% to print a literal %)
 
--- Integer division trap ---
7 / 2 = 3
double wrong = 3.00
double right = 3.50
 
--- Modulo with negatives (C99+) ---
7 / 2 = 3 7 % 2 = 1
-7 / 2 = -3 -7 % 2 = -1
7 / -2 = -3 7 % -2 = 1
-7 / -2 = 3 -7 % -2 = -1
 
Identity check: (-7 / 2) * 2 + (-7 % 2) = -7
⚠️
Annotation: double wrong = x / y; still yields 3.00 because the division is performed on two ints before the result is converted. The cast in (double)x / y promotes y too, so the division is done in floating point.

Example 2 — Relational operators

Every comparison produces the int value 0 or 1.

02_relational.c
#include <stdio.h>

int main(void)
{
    int a = 10, b = 20, c = 10;

    /* Every relational expression yields an int: 1 (true) or 0 (false) */
    printf("a = %d, b = %d, c = %d\n\n", a, b, c);
    printf("a == b : %d\n", a == b);
    printf("a == c : %d\n", a == c);
    printf("a != b : %d\n", a != b);
    printf("a <  b : %d\n", a <  b);
    printf("a >  b : %d\n", a >  b);
    printf("a <= c : %d\n", a <= c);
    printf("a >= b : %d\n", a >= b);

    /* The result is a plain int, not a special bool type */
    printf("\nsizeof(a < b)   = %zu bytes\n", sizeof(a < b));
    printf("(a < b) + (b > c) = %d\n", (a < b) + (b > c));

    /* Any non-zero value counts as true in a condition */
    int n = 42;
    if (n)
        printf("n = 42 is treated as true\n");

    /* '=' assigns, '==' compares: the result of an assignment is the assigned value */
    int x = 5;
    int r = (x = 0);               /* assigns 0 to x; r receives 0 */
    printf("\nAfter r = (x = 0): x = %d, r = %d\n", x, r);
    printf("x == 0 gives %d\n", x == 0);

    return 0;
}
▶
Output:
a = 10, b = 20, c = 10
 
a == b : 0
a == c : 1
a != b : 1
a < b : 1
a > b : 0
a <= c : 1
a >= b : 0
 
sizeof(a < b) = 4 bytes
(a < b) + (b > c) = 2
n = 42 is treated as true
 
After r = (x = 0): x = 0, r = 0
x == 0 gives 1
💡
Annotation: the size of a < b is the size of an int, proving the result is not a special boolean type. Also see how the value of an assignment expression (x = 0) is the value assigned — which is why if (x = 0) is a bug.

Example 3 — Short-circuit evaluation

Helper functions with printf side effects reveal exactly which operands are evaluated.

03_logical.c
#include <stdio.h>

/* Each helper announces itself, so we can see exactly which operands are evaluated */
static int left(int value)
{
    printf("  left()  evaluated -> %d\n", value);
    return value;
}

static int right(int value)
{
    printf("  right() evaluated -> %d\n", value);
    return value;
}

int main(void)
{
    int result;

    printf("1) left(0) && right(1)   [AND with false left]\n");
    result = left(0) && right(1);
    printf("  result = %d   (right() was skipped)\n\n", result);

    printf("2) left(1) && right(1)   [AND with true left]\n");
    result = left(1) && right(1);
    printf("  result = %d   (both evaluated)\n\n", result);

    printf("3) left(1) || right(0)   [OR with true left]\n");
    result = left(1) || right(0);
    printf("  result = %d   (right() was skipped)\n\n", result);

    printf("4) left(0) || right(0)   [OR with false left]\n");
    result = left(0) || right(0);
    printf("  result = %d   (both evaluated)\n\n", result);

    /* Practical use: the guard on the left protects the right-hand side */
    int divisor = 0;
    int dividend = 10;
    printf("5) divisor != 0 && dividend / divisor > 1\n");
    if (divisor != 0 && dividend / divisor > 1)
        printf("  quotient is greater than 1\n");
    else
        printf("  guard stopped the division by zero\n");

    return 0;
}
▶
Output:
1) left(0) && right(1) [AND with false left]
left() evaluated -> 0
result = 0 (right() was skipped)
 
2) left(1) && right(1) [AND with true left]
left() evaluated -> 1
right() evaluated -> 1
result = 1 (both evaluated)
 
3) left(1) || right(0) [OR with true left]
left() evaluated -> 1
result = 1 (right() was skipped)
 
4) left(0) || right(0) [OR with false left]
left() evaluated -> 0
right() evaluated -> 0
result = 0 (both evaluated)
 
5) divisor != 0 && dividend / divisor > 1
guard stopped the division by zero
💡
Annotation: in cases 1 and 3 the message from right() never appears — it was not called. Case 5 shows the practical payoff: the left-hand test guards the division.

Example 4 — Bit flags: set, clear, toggle, test

Uses a print_bin helper that loops over the bits from 7 down to 0 with (value >> bit) & 1.

04_bitwise.c
#include <stdio.h>

#define FLAG_READ   (1u << 0)   /* 0000 0001 */
#define FLAG_WRITE  (1u << 1)   /* 0000 0010 */
#define FLAG_EXEC   (1u << 2)   /* 0000 0100 */
#define FLAG_HIDDEN (1u << 3)   /* 0000 1000 */

/* Print an 8-bit value as binary, most significant bit first */
static void print_bin(const char *label, unsigned char value)
{
    printf("%-22s", label);
    for (int bit = 7; bit >= 0; bit--) {
        putchar(((value >> bit) & 1u) ? '1' : '0');
    }
    printf("  (decimal %u)\n", (unsigned)value);
}

int main(void)
{
    unsigned char flags = 0;
    print_bin("start:", flags);

    /* SET: OR with the mask turns the bit on */
    flags |= FLAG_READ;
    flags |= FLAG_EXEC;
    print_bin("set READ and EXEC:", flags);

    /* TEST: AND with the mask isolates the bit */
    printf("READ  set? %s\n", (flags & FLAG_READ)  ? "yes" : "no");
    printf("WRITE set? %s\n", (flags & FLAG_WRITE) ? "yes" : "no");

    /* TOGGLE: XOR with the mask flips the bit */
    flags ^= FLAG_WRITE;
    print_bin("toggle WRITE:", flags);
    flags ^= FLAG_WRITE;
    print_bin("toggle WRITE again:", flags);

    /* CLEAR: AND with the inverted mask turns the bit off */
    flags &= ~FLAG_EXEC;
    print_bin("clear EXEC:", flags);

    /* Shifts: multiply and divide by powers of two */
    unsigned char v = 5;
    print_bin("v = 5:", v);
    print_bin("v << 2 (5 * 4):", (unsigned char)(v << 2));
    print_bin("v >> 1 (5 / 2):", (unsigned char)(v >> 1));

    return 0;
}
▶
Output:
start: 00000000 (decimal 0)
set READ and EXEC: 00000101 (decimal 5)
READ set? yes
WRITE set? no
toggle WRITE: 00000111 (decimal 7)
toggle WRITE again: 00000101 (decimal 5)
clear EXEC: 00000001 (decimal 1)
v = 5: 00000101 (decimal 5)
v << 2 (5 * 4): 00010100 (decimal 20)
v >> 1 (5 / 2): 00000010 (decimal 2)
💡
Annotation: set = |= mask; clear = &= ~mask; toggle = ^= mask; test = & mask. The casts back to unsigned char are needed because the shift result is promoted to int.

Example 5 — Operator precedence

Expressions that look ambiguous, shown next to their explicitly parenthesised equivalents.

05_precedence.c
#include <stdio.h>

int main(void)
{
    int a = 1, b = 0, c = 0;

    printf("%-26s %-26s\n", "Without parentheses", "With explicit parentheses");
    printf("%-26s %-26s\n", "-------------------", "-------------------------");

    /* * binds tighter than + */
    printf("2 + 3 * 4        = %-6d (2 + 3) * 4      = %d\n", 2 + 3 * 4, (2 + 3) * 4);

    /* - is left-associative */
    printf("10 - 4 - 3       = %-6d 10 - (4 - 3)     = %d\n", 10 - 4 - 3, 10 - (4 - 3));

    /* / and * have equal precedence: left to right */
    printf("100 / 10 * 5     = %-6d 100 / (10 * 5)   = %d\n", 100 / 10 * 5, 100 / (10 * 5));

    /* + binds tighter than << */
    printf("1 + 2 << 3       = %-6d 1 + (2 << 3)     = %d\n", 1 + 2 << 3, 1 + (2 << 3));

    /* == binds tighter than & : classic bug */
    printf("6 & 2 == 2       = %-6d (6 & 2) == 2     = %d\n", 6 & 2 == 2, (6 & 2) == 2);

    /* && binds tighter than || */
    printf("a || b && c      = %-6d (a || b) && c    = %d\n", a || b && c, (a || b) && c);

    /* assignment is right-associative */
    int x, y, z;
    x = y = z = 7;
    printf("\nx = y = z = 7    -> x=%d y=%d z=%d (parsed as x = (y = (z = 7)))\n", x, y, z);

    return 0;
}
▶
Output:
Without parentheses With explicit parentheses
------------------- -------------------------
2 + 3 * 4 = 14 (2 + 3) * 4 = 20
10 - 4 - 3 = 3 10 - (4 - 3) = 9
100 / 10 * 5 = 50 100 / (10 * 5) = 2
1 + 2 << 3 = 24 1 + (2 << 3) = 17
6 & 2 == 2 = 0 (6 & 2) == 2 = 1
a || b && c = 1 (a || b) && c = 0
 
x = y = z = 7 -> x=7 y=7 z=7 (parsed as x = (y = (z = 7)))
⚠️
Annotation: compilers warn about several lines (-Wparentheses, -Wshift-op-parentheses, -Wlogical-op-parentheses) — that is the compiler telling you to add the parentheses shown in the right-hand column. Note 6 & 2 == 2 is 6 & (2 == 2) = 6 & 1 = 0, almost never what you meant.

Example 6 — The ternary operator

Min, max, absolute value, a nested ternary, and a demonstration that only one branch is evaluated.

06_ternary.c
#include <stdio.h>

static int min_of(int a, int b) { return a < b ? a : b; }
static int max_of(int a, int b) { return a > b ? a : b; }
static int abs_of(int n)        { return n < 0 ? -n : n; }

int main(void)
{
    int a = 12, b = -7;

    printf("min(%d, %d) = %d\n", a, b, min_of(a, b));
    printf("max(%d, %d) = %d\n", a, b, max_of(a, b));
    printf("abs(%d)    = %d\n", b, abs_of(b));

    /* Ternary as an expression inside printf */
    for (int n = 1; n <= 4; n++)
        printf("%d is %s\n", n, n % 2 == 0 ? "even" : "odd");

    /* Nested ternary: sign of a number (right-associative chain) */
    int values[] = { -5, 0, 9 };
    for (int i = 0; i < 3; i++) {
        int n = values[i];
        const char *sign = n < 0 ? "negative"
                         : n == 0 ? "zero"
                         : "positive";
        printf("%2d is %s\n", n, sign);
    }

    /* Only the chosen branch is evaluated */
    int hits = 0;
    int r = (a > 0) ? ++hits : (hits += 100);
    printf("r = %d, hits = %d (the else branch never ran)\n", r, hits);

    return 0;
}
▶
Output:
min(12, -7) = -7
max(12, -7) = 12
abs(-7) = 7
1 is odd
2 is even
3 is odd
4 is even
-5 is negative
0 is zero
9 is positive
r = 1, hits = 1 (the else branch never ran)
⚠️
Readability warning: the nested ternary above is about the limit of what is comfortable, and only because it is laid out one condition per line. Deeper nesting, or ternaries mixed with side effects, are better written as if/else.
